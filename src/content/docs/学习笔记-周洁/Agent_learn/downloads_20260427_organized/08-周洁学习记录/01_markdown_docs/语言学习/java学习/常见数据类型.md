## ArrayList

import java.util.ArrayList;  // 引入 ArrayList 类
ArrayList<String> sites = new ArrayList<String>();  // 初始化

for (String i : sites) {  // 使用 for-each 迭代元素
    System.out.println(i);
}
// 排序
import java.util.Collections;  // 引入 Collections 类

## HashMap
```
import java.util.HashMap;  // 引入 HashMap 类
HashMap<Integer, String> Sites = new HashMap<Integer, String>();  // 初始化
Sites.put(1, "Google");  // 添加元素
Sites.get(1);  // 访问元素
Sites.remove(1);  // 删除元素
Sites.clear();  // 删除所有元素
Sites.size();  // 元素数量


for (Integer i : Sites.keySet()) {  // 输出 key 和 value
    System.out.println("key: " + i + " value: " + Sites.get(i));
}
Sites.keySet();  // key 组成的 set 集合
Sites.values();  // value 组成的 Set 集合
Sites.containsKey(1); // 是否存在指定的 key 
Sites.containsValue("Google"); // 是否存在指定的 value 
String value = Sites.replace(1, "Wiki");  // 返回 Google
Sites.replace(1, "Google", "Wiki");  // 返回 true
Sites.replaceAll((key, value) -> value.toUpperCase());  // 将所有的值更改为大写
Sites.getOrDefault(1, "Not Found");  // 如果找不到 key ，则返回设置的默认值
Sites.forEach((key, value) -> {
    System.out.print(key + "=" + value + " ");
});
import java.util.Map.Entry;
for(Entry<Integer, String> entry: Sites.entrySet()) {
    System.out.print(entry);
}
prices.put("Shoes", 200);
// 如果 key 对应的 value 不存在，则返回该 value 值，如果存在，则返回通过 remappingFunction 重新计算后的值。
int returnedValue = prices.merge("Shirt", 100, (oldValue, newValue) -> oldValue + newValue);
```
## Iterator
```
import java.util.ArrayList;
import java.util.Iterator;  // 引入 Iterator 类
ArrayList<String> sites = new ArrayList<String>();
sites.add("Google");
Iterator<String> it = sites.iterator();  // 获取迭代器
System.out.println(it.next());  // 输出集合中的第一个元素
while(it.hasNext()) {  // 循环集合元素
    System.out.println(it.next());
}
```
