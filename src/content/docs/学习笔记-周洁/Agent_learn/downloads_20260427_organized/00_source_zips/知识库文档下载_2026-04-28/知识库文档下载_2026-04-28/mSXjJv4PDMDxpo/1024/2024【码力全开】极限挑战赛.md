import pygame as g,sys
W,H,S,N=640,640,32,20
B=(0,0,0)
d=[[0]*N for _ in range(N)]
g.init()
s=g.display.set_mode((W,H))
r,p=1,1
def w(x,y,p):
 for a,b in[(1,0),(0,1),(1,1),(1,-1)]:
  l=r=0
  for i in range(1,5):
   if 0<=x+a*i<N and 0<=y+b*i<N and d[x+a*i][y+b*i]==p:r+=1
   if i>0 and 0<=x-a*i<N and 0<=y-b*i<N and d[x-a*i][y-b*i]==p:l+=1
   if l+r+1>=5:return 1
 return 0
while r:
 for e in g.event.get():
  if e.type==g.QUIT:r=0
  elif e.type==g.MOUSEBUTTONDOWN:
   x,y=e.pos[0]//S,e.pos[1]//S
   if d[x][y]==0:
    d[x][y]=p
    if w(x,y,p):print('p%s win!' % p);r=0
    p=3-p
 s.fill('white')
 for i in range(N):
  l=i*S;g.draw.line(s,B,(0,l),(W,l),1);g.draw.line(s,B,(l,0),(l,H),1)
 for x in range(N):
  for y in range(N):
   if d[x][y]:
    h=S//2;g.draw.circle(s,B,(x*S+h,y*S+h),h if d[x][y]%2 else h,2if d[x][y]%2==0 else 0)
 g.display.flip()
for i in d:print(*i)
g.quit();sys.exit()
