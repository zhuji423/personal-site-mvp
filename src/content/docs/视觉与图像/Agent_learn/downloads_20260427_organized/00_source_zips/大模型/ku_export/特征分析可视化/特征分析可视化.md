# 特征分析可视化
import pandas as pdfrom sklearn.model_selection import train_test_splitimport matplotlib.pyplot as pltimport seaborn as snsimport ydata_profiling as pp # 导入数据集iris_local = pd.read_csv("./data/dataset/iris.csv", usecols=[0, 1, 2, 3, 4])  # 选择特征列、标签列iris_local = iris_local.dropna()  # 丢弃含空值的行、列print(iris_local.head())  # 查看前几行# 看下特征数据分布print(iris_local.describe())![](https://rte.weiyun.baidu.com/wiki/attach/image/api/imageDownloadAddress?attachId=adc24efde84041e291049706440ea7d9&docGuid=0mr9CT8W9ZtHLq)

# 箱线图iris_local.plot(kind='box', subplots=True, layout=(2, 3), sharex=False, sharey=False)# 直方图iris_local.hist()# 分布图iris_local.plot()# kde核密度分布图iris_local.plot(kind='kde')# 径向可视化plt.figure()ax = pd.plotting.radviz(iris_local, 'res_real', colormap='brg')ax.add_artist(plt.Circle((0, 0), 1, color='r', fill=False))plt.show()# 生成各特征之间关系的矩阵图plt.figure()# 设置颜色主题antV = ['#1890FF', '#2FC25B', '#FACC14', '#223273', '#8543E0', '#13C2C2', '#3436c7', '#F04864']sns.pairplot(data=iris_local, palette=antV, hue='res_real')plt.show()![](https://rte.weiyun.baidu.com/wiki/attach/image/api/imageDownloadAddress?attachId=957794620d59478784bad67fe9c19d8b&docGuid=0mr9CT8W9ZtHLq)

# 使用 Andrews Curves 将每个多变量观测值转换为曲线并表示傅立叶级数的系数，这对于检测时间序列数据中的异常值很有用。plt.figure()pd.plotting.andrews_curves(iris_local, 'res_real', colormap='cool')plt.show()# 平行坐标plt.figure()pd.plotting.parallel_coordinates(iris_local, 'res_real', colormap='brg')plt.show()# 通过热图找出数据集中不同特征之间的相关性，高正值或负值表明特征具有高度相关性plt.figure()sns.heatmap(iris_local.corr(), annot=True, cmap='GnBu', linewidths=1, linecolor='k',                  square=True, mask=False, vmin=-1, vmax=1, cbar_kws={"orientation": "vertical"}, cbar=True)plt.show()![](https://rte.weiyun.baidu.com/wiki/attach/image/api/imageDownloadAddress?attachId=c5a471db1124441e8a594d3dad82edf2&docGuid=0mr9CT8W9ZtHLq)
