自动更新工具执行结果

# 1、基于SDK (推荐)
```
# !/usr/bin/python3

# -*- coding: utf-8 -*-

"""
@author: GuYouda
@date:   2024/2/27
@desc:
"""
import logging
import json
import requests
from qep_delivery_component import UrlInlaidComponent
from qep_delivery_data import init_log

from qep_delivery_tools import ToolResult
from qep_delivery_tools import ToolResultCallbackRequest
from qep_delivery_tools import ToolExecuteStatusEnum

# TODO 替换成自己的账号
TESTONE_USERNAME = ''
TESTONE_TOKEN = ''


def generate_parameter() -> ToolResultCallbackRequest:
    """
    生成请求参数
    :return: 生成的请求参数
    """
    param = ToolResultCallbackRequest(
        engine_task_id='d9008d80-0636-4e4c-8707-717ba4918c60',  # TODO 替换成需要更新的任务id
        tool_results=[
            ToolResult(
                tool_id="ebda091aa38bcea2333c6da72e9fb795",  # TODO 替换成需要更新的工具id
                component_info=UrlInlaidComponent(  # TODO 替换成自己的工具的结果组件信息
                    component_operate_config=None,
                    component_data="https://aisuda.baidu-int.com/home"
                ),
                tool_status=ToolExecuteStatusEnum.RUNNING,  # TODO 替换成自己的工具状态
                tool_message="工具执行中",  # TODO 替换成自己的工具状态信息提示
                high_risk=False,  # TODO 替换成自己的工具风险状态
                risk_summary=""  # TODO 替换成自己的工具风险信息摘要
            )
        ]
    )

    return param


def update(param: ToolResultCallbackRequest) -> bool:
    """
    更新结果
    :param param: 请求参数
    :return: 是否成功
    """
    if not TESTONE_USERNAME or not TESTONE_TOKEN:
        raise Exception("username token 不能为空")

    headers = {
        "Content-Type": "application/json",
        "username": TESTONE_USERNAME,
        "token": TESTONE_TOKEN,
    }
    url = "https://testone.baidu-int.com/engine/storyAgent/updateToolExecuteResult"
    
    try:
        param = param.to_json(True)
        logging.info("tool result callback, url:[%s], parameter:[%s]", url, param)    
        res = requests.post(url, headers=headers, data=param.encode("utf-8"))
        logging.info("tool result callback, response:[%s]", res.text)
        content = json.loads(res.content)
    except Exception as e:
        logging.error("response parse error: %s", e)
        content = None
    if content is None or content["code"] != 0:
        return False
    return True


def main():
    init_log()
    param = generate_parameter()
    update_result = update(param)
    logging.info("update result: %s", update_result)


if __name__ == '__main__':
    main()

```


# 2、自己调用接口
接口地址：https://testone.baidu-int.com/engine/storyAgent/updateToolExecuteResult

请求类型：POST

请求头：

ContentType:application/json

username: 用户名

token: 自己在TestOne平台的token  (获取方式：[https://testone.baidu-int.com/api/getToken](https://testone.baidu-int.com/api/getToken))

请求参数：

|参数名|参数类型|是否必填|含义|
|-|-|-|-|
|engineTaskId|string|是|平台Agent调度引擎生成的唯一任务ID，在调用Agent的时候会给到这个参数|
|toolResults|array[Object]|是|需要更新的工具的结果列表|
| >toolId|string|是|工具ID 需要更新哪个工具的结果|
| >toolStatus|int|是|工具的状态 （只有在执行中的工具结果允许被更新，如果希望后续持续更新，需要设置为执行中）0：成功1：失败2：执行中|
| >toolMessage|string|否|状态解释，主要是异常的时候使用|
| >highRisk|bool|是|工具是否高风险|
| >riskSummary|string|是|工具的风险摘要|
| >componentInfo|object|是|工具结果前端渲染使用的组件信息|
|    >>componentId|int|是|组件ID|
|    >>componentOperateConfig|object|是|组件的操作配置|
|        >>>expand|bool|是|是否默认展开|
|        >>>enableRefresh|bool|是|是否允许刷新（重新触发）|
|    >>componentData|anyType|是|具体的组件数据，支持任意类型。具体每个组件的数据结构详见 [组件设计](https://ku.baidu-int.com/knowledge/HFVrC7hq1Q/3LoFUu-M_a/U-RwwdZ7ut/AZghNooBQFtzZm?t=mention&mt=doc&dt=doc)|
