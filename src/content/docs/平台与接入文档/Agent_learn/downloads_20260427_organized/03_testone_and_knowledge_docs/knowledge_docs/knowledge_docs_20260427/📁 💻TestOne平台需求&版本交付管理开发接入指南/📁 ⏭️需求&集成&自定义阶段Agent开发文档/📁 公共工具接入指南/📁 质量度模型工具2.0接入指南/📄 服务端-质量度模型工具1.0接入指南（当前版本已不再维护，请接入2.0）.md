服务端-质量度模型工具1.0接入指南（当前版本已不再维护，请接入2.0）

# 1、背景
为了支持之前在QTEST提测的业务，能够切换到TestOne提测，并且在TestOne测试阶段调起之前在QTEST已有的质量度模型



# 2、接入指南
> 以下方案二选一，建议2.1  
> 针对部分业务调起质量度模型还需要其他更多参数的场景，建议2.2

## 2.2、业务自行调用
> 如果业务存在调起石雪提供的质量度模型的时候，还需要传递更多参数，业务参考以下代码自己改造即可
代码Demo

```
# !/usr/bin/python3

# -*- coding: utf-8 -*-

"""
@author: GuYouda
@date:   2024/3/19
@desc:
"""
import json
import logging
from datetime import datetime, timedelta
from typing import Dict, List

from qep_delivery_component import ComponentOperateConfig
from qep_delivery_component import DefaultRiskComponent
from qep_delivery_component import RiskData
from qep_delivery_data import ICafeUtils
from qep_delivery_data import KirinUtils
from qep_delivery_data import QadcUtils
from qep_delivery_data import split_icafe_card_to_space_and_sequence
from qep_delivery_tools.common.entity.parameter import Parameter
from qep_delivery_tools.common.entity.tool_result import ToolResult
from qep_delivery_tools.requirement_agent.base_tool import BaseTool


class XxxTool(BaseTool):
    """
    服务端质量度风险模型工具
    麒麟策略ID：211564
    """

    def __init__(self, testone_username: str, testone_token: str, kirin_username: str, kirin_token: str,
                 qadc_username: str, qadc_token: str, icafe_username: str, icafe_password: str, service_id: int):
        """
        初始化
        :param testone_username: TestOne平台用户名
        :param testone_token: Testone平台用户对应的API访问Token
        :param kirin_username: 麒麟用户名
        :param kirin_token: 麒麟Token
        :param qadc_username: QADC平台用户名
        :param qadc_token: QADC平台对应的token
        :param icafe_username: icafe平台的用户名
        :param icafe_password: icafe平台的访问密码,
            获取方式：https://ku.baidu-int.com/knowledge/HFVrC7hq1Q/_SKPgSwp2G/NbX2gitgSF/4cc48b299a6846
        :param service_id: 自己需求对应的业务在QTEST平台配置的服务标识ID 获取方式：QTEST->左侧导航栏->质量度特征配置
        """
        super().__init__("aee740c7f2226846eccf2ad28647ed00")
        self.kirin_strategy_id = 211564
        self.testone_username = self.validate_str_not_blank(testone_username)
        self.testone_token = self.validate_str_not_blank(testone_token)
        self.kirin_username = self.validate_str_not_blank(kirin_username)
        self.kirin_token = self.validate_str_not_blank(kirin_token)
        self.qadc_username = self.validate_str_not_blank(qadc_username)
        self.qadc_token = self.validate_str_not_blank(qadc_token)
        self.icafe_username = self.validate_str_not_blank(icafe_username)
        self.icafe_password = self.validate_str_not_blank(icafe_password)

        self.service_id = service_id

    def get_param_from_qadc(self, parameter: Parameter) -> List[Dict]:
        """
        从QADC获取请求参数
        """
        # 获取本月第一天时间
        f_d_this_month = datetime.today().replace(day=1)
        # 获取上月最后一天时间
        l_d_last_month = f_d_this_month - timedelta(days=1)

        # 需要查询的分表: 本月，上一月
        tables = [
            f_d_this_month.strftime("rd_tice_info_%Y%m"),
            l_d_last_month.strftime("rd_tice_info_%Y%m")
        ]

        sql = ("SELECT story_id, module_info, pipeline_build_id FROM {} where belong='testone' "
               "AND story_id in ('{}') ORDER BY @timestamp DESC").format(
            ",".join(tables), "', '".join(parameter.icafe_ids))
        module_list = []
        try:
            response = QadcUtils(self.qadc_username, self.qadc_token).get_result_from_qadc(
                table="rd_tice_info", sql=sql)
            if not response.data:
                return module_list
            for item in response.data:
                module_info = json.loads(item['module_info'])
                for i in module_info:
                    module_list.append(
                        {
                            "module": i.get("name"),
                            "branch": i.get("branch"),
                            "commitId": i.get("commitId")
                        }
                    )
        except Exception as e:
            logging.error(e)

        return module_list

    def get_param_from_icafe(self, parameter: Parameter) -> List[Dict]:
        """
        从icafe卡片获取请求参数
        """

        module_list = []
        module_info = {}
        for icafe_id in parameter.icafe_ids:
            card_space, card_id = split_icafe_card_to_space_and_sequence(icafe_id)
            try:
                response = ICafeUtils(self.icafe_username, self.icafe_password).get_code_dev_info(card_space, card_id)
                if not response.codeMerge:
                    continue

                for item in response.codeMerge:
                    if item.projectName not in module_info:
                        module_info[item.projectName] = {}
                    # 取当前分支最新commitId
                    if item.branch not in module_info[item.projectName]:
                        module_info[item.projectName][item.branch] = item.gitCommits[0].commitId
                        module_list.append(
                            {
                                "module": item.projectName,
                                "branch": item.branch,
                                "commitId": item.gitCommits[0].commitId
                            }
                        )
            except Exception as e:
                logging.error(e)
        return module_list

    def execute(self, parameter: Parameter) -> ToolResult:
        """
        执行
        """
        # QTEST查询参数
        module_info = self.get_param_from_qadc(parameter)
        # 如果从QTEST没有查到代码模块参数，从iCafe查询
        if not module_info:
            module_info = self.get_param_from_icafe(parameter)

        if not module_info:
            logging.warning("未获取到卡片关联的代码信息，不支持调用服务端质量度模型")
            component_info = DefaultRiskComponent(
                ComponentOperateConfig(expand=True, enable_refresh=False),
                RiskData(parameter.kirin_task_id, False, "未获取到卡片关联的代码信息，不支持调用服务端质量度模型",
                         "未获取到卡片关联的代码信息，不支持调用服务端质量度模型", 100, "")
            )

            # 返回工具执行结果
            return ToolResult(self.tool_id, component_info, high_risk=False,
                              risk_summary="未获取到卡片关联的代码信息，不支持调用服务端质量度模型")

        kirin_task_id = 0
        score = 0
        url = ""
        try:
            strategy_param = {
                "story_id": ",".join(parameter.icafe_ids),
                "moduleInfo": module_info,
                "service_id": self.service_id
            }
            kirin_response = KirinUtils(self.kirin_username, self.kirin_token).call_kirin_api(
                self.kirin_strategy_id, json.dumps(strategy_param, ensure_ascii=False))
            kirin_task_id = kirin_response.taskId
            strategy_response = json.loads(kirin_response.response)

            high_risk = strategy_response.get("is_exemption_predict", 0) == 0  # 0高风险 1低风险
            summary = "质量度模型结果为高风险" if high_risk else "质量度模型结果为低风险"
            detail = strategy_response.get("reason", "")
            score = round(float(strategy_response.get("is_exemption_predict_score", 0)) * 100, 2)
            url = strategy_response.get("report_url", "")
        except Exception as e:
            logging.error(e)
            high_risk = True
            summary = "质量度模型调用异常，默认结果为高风险"
            detail = "质量度模型调用异常，默认结果为高风险"

        risk_component_data = RiskData(kirin_task_id, high_risk, summary, detail, score, url)

        # 构建前端风险组件
        component_info = DefaultRiskComponent(
            ComponentOperateConfig(expand=True, enable_refresh=False), risk_component_data)

        # 返回工具执行结果
        return ToolResult(self.tool_id, component_info, high_risk=risk_component_data.high_risk,
                          risk_summary=risk_component_data.risk_summary)
```
