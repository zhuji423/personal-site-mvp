```
curl --location 'https://testone.baidu-int.com/engine/storyAgent/executeResult/getToolExecuteHistoryByIcafeAndStage' \
--header 'username: NAME' \
--header 'token: TOKEN' \
--header 'Content-Type: application/json' \
--data '{
    "stage": 4,
    "icafeId": "FEEDADS-6343",
    "toolIdList": [
        "c5e23f1a22974a1717d1889359d80ca2"
    ]
}'
```


* Python版 （ChatGPT实现）

> 注意替换参数，异常处理
```
import requests

def get_tool_execute_history(username, token, stage, icafe_id, tool_id_list):
    url = 'https://testone.baidu-int.com/engine/storyAgent/executeResult/getToolExecuteHistoryByIcafeAndStage'
    headers = {
        'username': username,
        'token': token,
        'Content-Type': 'application/json'
    }
    data = {
        'stage': stage,
        'icafeId': icafe_id,
        'toolIdList': tool_id_list
    }
    response = requests.post(url, headers=headers, json=data)
    return response.json()

# Example usage:
username = 'NAME'
token = 'TOKEN'
stage = 4
icafe_id = 'FEEDADS-6343'
tool_id_list = ['c5e23f1a22974a1717d1889359d80ca2']
result = get_tool_execute_history(username, token, stage, icafe_id, tool_id_list)
print(result)
```
