风险评分组件

# 示例代码
```
from qep_delivery_component import ComponentOperateConfig
from qep_delivery_component import RiskData
from qep_delivery_component import DefaultRiskComponent
from qep_delivery_tools import Parameter
from qep_delivery_tools import ToolResult
from qep_delivery_tools import BaseTool


class XxxTool(BaseTool):
    """
    Xxx工具
    """

    def __init__(self):
        super().__init__(自己注册的ToolId)


    def execute(self, parameter: Parameter) -> ToolResult:
        """
        内置风险展示组件
        """
        kirin_task_id = 123456789  # 麒麟平台任务ID，方便策略方自己排查问题
        detail = """
        自测数据不达标，<font color='red' style="font-weight:bold;">提测准入不通过</font> 如有疑问，请联系 <a href='baidu://message?id=GuYouda'>guyouda <img src='https://eefe.baidu-int.com/avatars/guyouda' style='width: 18px; height: 18px; border-radius: 50%; vertical-align: middle'></a><br>
        附：<a href='https://ku.baidu-int.com/d/Dr7Vtk8Zt_saQ-'>提测准入流程规范</a><br>
        <table border="1" cellspacing="0" bordercolor="gray">
            <thead><col style='width: 15%;text-align: center;'>
                <col style='width: 10%;text-align: center;'>
                <tr>
                    <th>检查项</th>
                    <th>是否通过</th>
                    <th>备注</th>
                </tr>
                </thead>
            <tbody>
                <tr>
                    <td>自测case签章率</td>
                    <td><font color='red'>不通过</font></td>
                    <td>自测case签章率不达标，签章数据为:40.0%(10/25)。</td>
                </tr>
                <tr>
                    <td>自测函数覆盖率</td>
                    <td><font color='green'>通过</font></td>
                    <td>自测函数覆盖率达标，覆盖率数据为:100.0%(50/50)</td>
                </tr>
            </tbody>
        </table>	
        """
        high_risk = True
        score = 38.8
        summary = "提测准入不通过"
        url = ""  # 如果有第三方报告地址，支持超链跳转第三方
    
        risk_data = RiskData(kirin_task_id, high_risk, summary, detail, score. url)
        component_info = DefaultRiskComponent(component_operate_config=ComponentOperateConfig(), component_data=risk_data)
        return ToolResult(self.tool_id, component_info)

```
