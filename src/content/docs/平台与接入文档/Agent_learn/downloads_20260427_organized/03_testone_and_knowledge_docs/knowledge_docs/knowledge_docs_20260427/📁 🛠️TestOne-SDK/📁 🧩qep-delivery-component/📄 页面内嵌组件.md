页面内嵌组件

# 示例代码
注意：仅支持内嵌 https 协议的网页

```
from qep_delivery_component import ComponentOperateConfig
from qep_delivery_component import UrlInfo
from qep_delivery_component import UrlInlaidComponent
from qep_delivery_tools import Parameter
from qep_delivery_tools import ToolResult
from qep_delivery_tools import BaseTool


class XxxTool(BaseTool):
    """
    Xxx工具
    """

    def __init__(self):
        super().__init__(自己注册的ToolId)


    def execute(self, parameter: Parameter) -> ToolResult:
        """
        URL内嵌展示组件
        """
        # 需要内嵌展示的URL（支持内嵌多个）
        url_list = [
            UrlInfo(
                name="流水线信息", 
                url="https://console.cloud.baidu-int.com/devops/ipipe/workspaces/175037/pipeline-builds/162456162/view"
            ),
            UrlInfo(
                name="风险报告", 
                url="https://amis.baidu.com/group/quality_risk_report/index?product_id=232&relation_id=xiaochengxu00-64921&stage=2&relation_type=Story&embed=true&tag=ranker", 
                height=600  # 组件显示的高度（不赋值默认400px）
            )
        ]
        component_info = UrlInlaidComponent(component_operate_config=ComponentOperateConfig(), component_data=url_list)
        return ToolResult(self.tool_id, component_info)
```
