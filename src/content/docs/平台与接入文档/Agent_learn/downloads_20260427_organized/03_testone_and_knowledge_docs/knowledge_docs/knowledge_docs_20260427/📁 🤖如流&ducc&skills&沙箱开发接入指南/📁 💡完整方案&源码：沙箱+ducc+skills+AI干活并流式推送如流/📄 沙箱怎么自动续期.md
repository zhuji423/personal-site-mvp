沙箱怎么自动续期

自己启动一个定时任务，每天自动执行一次续期操作即可

```
# !/usr/bin/python3

# -*- coding: utf-8 -*-

"""
@author: GuYouda
@date:   2026/1/15
@desc:   沙箱续期
"""
import os
from datetime import datetime, timedelta

from e2b_code_interpreter import Sandbox


def is_sandbox_running(sbx: Sandbox):
    """
    判断沙箱是否正在运行
    :param sbx: 沙箱对象
    :return: True or False
    """
    return sbx and sbx.is_running()


def get_expire_time(sbx: Sandbox):
    """
    获取沙箱过期时间
    :param sbx: 沙箱对象
    :return: 沙箱过期时间
    """
    return sbx.get_info().end_at.astimezone()


def get_start_time(sbx: Sandbox):
    """
    获取沙箱启动时间
    :param sbx: 沙箱对象
    :return: 沙箱启动时间
    """
    return sbx.get_info().started_at.astimezone()


# 计算剩余时间
def get_remaining_time(sbx: Sandbox) -> int:
    """
    获取剩余时间
    :param sbx: 沙箱对象
    :return: 剩余时间，单位秒
    """
    return int((sbx.get_info().end_at.astimezone() - datetime.now().astimezone()).total_seconds())


def extend_timeout(sbx: Sandbox, minutes: int):
    """
    延长沙箱%s过期时间, 特别注意：仅在沙盒未过期的情况下有效，否则会报错
    :param sbx: 沙箱对象
    :param minutes: 分钟数
    """
    if not is_sandbox_running(sbx):
        raise Exception("沙盒已过期，无法延长过期时间")

    # 计算延长后的过期时间
    end_time = get_expire_time(sbx) + timedelta(minutes=minutes)
    seconds = int((end_time - get_start_time(sbx)).total_seconds())
    sbx.set_timeout(seconds)


def set_timeout(sbx: Sandbox, timeout_str: str):
    """
    设置沙箱过期时间
    :param sbx: 沙箱对象
    :param timeout_str: 过期时间, 格式为：2026-01-01 00:00:00
    """
    if not is_sandbox_running(sbx):
        raise Exception("沙盒已过期，无法设置沙箱过期时间")
    timeout = datetime.strptime(timeout_str, "%Y-%m-%d %H:%M:%S")
    # 过期时间不可小于启动时间
    if timeout < get_start_time(sbx):
        raise Exception("过期时间不能小于启动时间")

    seconds = int((timeout - get_start_time(sbx)).total_seconds())
    sbx.set_timeout(seconds)


def main():
    """
    主函数
    :return:
    """
    # 
    os.environ["E2B_API_KEY"] = "<KEY>"
    os.environ["E2B_DOMAIN"] = "agent-sandbox.baidu-int.com"
    sbx = Sandbox(sandbox_id="xxxxx")

    print(sbx.is_running())
    print(get_expire_time(sbx))
    print(get_start_time(sbx))
    print(get_remaining_time(sbx))
    # extend_timeout(sbx, 10)
    set_timeout(sbx, "2026-01-00 00:00:00")
    extend_timeout(sbx, 10)


if __name__ == '__main__':
    main()
```
