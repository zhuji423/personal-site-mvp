# -*- coding: utf-8 -*-

"""

from qep_delivery_component import ComponentOperateConfig
from qep_delivery_component import DatetimeFormDetail
from qep_delivery_component import DigitalFormDetail
from qep_delivery_component import InputTypeEnum
from qep_delivery_component import MultiSelectFormDetail
from qep_delivery_component import RichTextFormDetail
from qep_delivery_component import SingleSelectFormDetail
from qep_delivery_component import SubmitTestReportComponent
from qep_delivery_component import SubmitTestReportDetail
from qep_delivery_component import TableFormDetail
from qep_delivery_component import TableFormHeadConfig
from qep_delivery_component import TextFormDetail
from qep_delivery_data import split_icafe_card_to_space_and_sequence
from qep_delivery_tools import BaseTool
from qep_delivery_tools import Parameter
from qep_delivery_tools import ToolResult


class SubmitTestReportTool(BaseTool):
    """
    提测报告组件下发工具
    """

    def __init__(self):
        super().__init__(自己注册的业务自定义提测报告)

    def execute(self, parameter: Parameter) -> ToolResult:
        
        other_fields = [
            TextFormDetail(
                name="name",
                display_value="需求名",
                required=True,
                min_length=2,
                max_length=10,
                tip="请填写需求名"
            ),
            DigitalFormDetail(
                name="devDays",
                display_value="开发时长",
                required=True,
                min=1,
                max=30,
                default_value=1,
                tip="请填写需求开发时长，单位(天)"
            ),
            SingleSelectFormDetail(
                name="modifyDatabase",
                display_value="是否设计数据库变更",
                required=True,
                select_items=["是", "否"],
                default_value=["否"]
            ),
            MultiSelectFormDetail(
                name="effect",
                display_value="影响范围",
                required=True,
                select_items=["首页Feed流", "卡片落地页", "下单流程", "个人中心", "无其他影响"],
                default_value=["卡片落地页"]
            ),
            DatetimeFormDetail(
                name="onlineDate",
                display_value="上线时间",
                required=True,
                tip="请填写需求的上线时间"
            ),

            RichTextFormDetail(
                name="description",
                display_value="其他详细描述",
                required=False
            ),
            TableFormDetail(
                name="codeInfo",
                display_value="代码信息",
                head=[
                    TableFormHeadConfig(
                        data_index="repository",
                        title="代码库",
                        type=InputTypeEnum.TEXT,
                        tip="请填写改动代码库"
                    ),
                    TableFormHeadConfig(
                        data_index="branch",
                        title="分支",
                        type=InputTypeEnum.TEXT,
                        tip="请填写改动代码分支"
                    ),
                    TableFormHeadConfig(
                        data_index="cr",
                        title="CR",
                        type=InputTypeEnum.TEXT,
                        tip="请填写改动代码CR地址"
                    ),
                    TableFormHeadConfig(
                        data_index="onlineType",
                        title="上线方式",
                        type=InputTypeEnum.SINGLE_SELECT,
                        select_items=["Jarvis", "Opera", "CNAP"],
                        width=200,
                        tip="请填写该模块的上线方式"
                    )
                ],
                default_value=[
                    {
                        "repository": "testone-server",
                        "branch": "chenyineng01_dev",
                        "cr": ("https://console.cloud.baidu-int.com/devops/icode/repos/"
                               "baidu/ep-qa/testone-server/tree/master"),
                        "onlineType": ["Jarvis"]
                    },
                    {
                        "repository": "testone-web",
                        "branch": "nyx_dev",
                        "cr": ("https://console.cloud.baidu-int.com/devops/icode/repos/"
                               "baidu/ep-qa/testone-web/tree/master"),
                        "onlineType": ["CNAP"]
                    },
                ],
                required=True
            )
        ]

        component = SubmitTestReportComponent(
            component_operate_config=ComponentOperateConfig(expand=True),
            component_data=SubmitTestReportDetail(
                default_qa_list=[],
                default_rd_list=["guyouda", "wushaoshan"],
                self_test=False,
                available_receive_email_list=["wangshitao", "guyouda", "wushaoshan", "chenyineng01", "nieyuxin"],
                default_receive_email_list=["guyouda"],
                available_carbon_copy_email_list=["qep-testone", "wangshitao", "guyouda"],
                default_carbon_copy_email_list=["qep-testone"],
                other_fields=other_fields
            )
        )

        return ToolResult(tool_id=self.tool_id, component_info=component)
