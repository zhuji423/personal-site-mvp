from qep_delivery_component import ComponentOperateConfig
from qep_delivery_component import DefaultRiskComponent
from qep_delivery_component import TableColumnTypeEnum
from qep_delivery_component import TableInnerUrl
from qep_delivery_component import UrlInfo
from qep_delivery_component import ListComponent
from qep_delivery_component import MixedComponent
from qep_delivery_component import RichTextComponent
from qep_delivery_component import RiskData
from qep_delivery_component import SubComponent
from qep_delivery_component import TableComponent
from qep_delivery_component import TableHead
from qep_delivery_component import TableDetail
from qep_delivery_component import UrlInlaidComponent
from qep_delivery_tools import BaseTool
from qep_delivery_tools import Parameter
from qep_delivery_tools import ToolResult


class XxxTool(BaseTool):
    """
    Xxx工具
    """

    def __init__(self):
        super().__init__(自己注册的ToolId)

    def execute(self, parameter: Parameter) -> ToolResult:
        """
        内置风险展示组件
        """
        # 富文本
        sub_component_1 = SubComponent(
            sub_title="子组件xxx",
            component=RichTextComponent(
                component_operate_config=None,
                component_data="这是一个富文本组件</br> <font color='red'>红色的文字; </font> <font>超链接：</font>"
                               "<a href='https://console.cloud.baidu-int.com' target='_blank'>点我打开</a>"
            )
        )

        # 表格
        sub_component_2 = SubComponent(
            sub_title="子组件123",
            component=TableComponent(
                component_operate_config=None,
                component_data=TableDetail(
                    head=[
                        TableHead(data_index="kirin_task_id", title="麒麟任务ID", width=100),
                        TableHead(data_index="engine_task_id", title="引擎任务ID", type=TableColumnTypeEnum.TEXT,
                                  width=150),
                        TableHead(data_index="icafe_rich_text", title="富文本", type=TableColumnTypeEnum.RICH_TEXT,
                                  width=300),
                        TableHead(data_index="icafe_url_blank", title="超链-新页面", type=TableColumnTypeEnum.URL),
                        TableHead(data_index="icafe_url_popup", title="超链-弹窗", type=TableColumnTypeEnum.URL),
                    ],
                    body=[
                        {
                            "kirin_task_id": 12345,
                            "engine_task_id": "xxx-xxx",
                            "icafe_rich_text": "<font color='green'>绿色的文字</font>",
                            "icafe_url_blank": TableInnerUrl(
                                name="链接-新页面打开",
                                url='https://console.cloud.baidu-int.com',
                                target='_blank'
                            ),
                            "icafe_url_popup": TableInnerUrl(
                                name="链接-浮层打开",
                                url='https://console.cloud.baidu-int.com',
                                target='_popup'
                            ),

                        },
                        {
                            "kirin_task_id": 56789,
                            "engine_task_id": "yyy-yyy",
                            "icafe_rich_text": "<font color='red'>红色的文字;  </font> <font>超链接:</font>"
                                               "<a href='https://console.cloud.baidu-int.com' target='_blank'>点我打开</a>",
                            "icafe_url_blank": TableInnerUrl(
                                name="链接-新页面打开",
                                url='https://console.cloud.baidu-int.com',
                                target='_blank'
                            ),
                            "icafe_url_popup": TableInnerUrl(
                                name="链接-浮层打开",
                                url='https://console.cloud.baidu-int.com',
                                target='_popup'
                            ),
                        }
                    ])
            )
        )

        # 列表
        sub_component_3 = SubComponent(
            sub_title="子组件ahjsgd",
            component=ListComponent(
                component_operate_config=ComponentOperateConfig(expand=False),
                component_data=["列表1", "列表2"]
            )
        )

        # URL
        sub_component_4 = SubComponent(
            sub_title="子组件hhhh",
            component=UrlInlaidComponent(
                component_operate_config=None,
                component_data=[
                    UrlInfo(name="链接A", url="https://ku.baidu-int.com/d/NzwrklmOjkeBeQ"),
                    UrlInfo(name="链接B", url="https://ku.baidu-int.com/d/NzwrklmOjkeBeQ"),
                ]
            )
        )

        # 风险组件
        sub_component_5 = SubComponent(
            sub_title="子组件fck",
            component=DefaultRiskComponent(
                component_operate_config=None,
                component_data=RiskData(12345, True, "风险概要", "风险详情", 59, "http://testone.baidu-int.com")
            ))
        component_info = MixedComponent(
            component_operate_config=None,
            component_data=[
                sub_component_1, sub_component_2, sub_component_3, sub_component_4, sub_component_5
            ]
        )
        return ToolResult(self.tool_id, component_info)
