# 2、【群聊】发送自定义消息到群聊
功能：

发送自定义消息到如流群

版本：

>=0.4.0

需要权限：

否

输入：

机器人 token  群ID   消息内容

输出：

是否发送成功

```
from qep_delivery_data import AtUserMsg
from qep_delivery_data import InfoFlowMsgContent
from qep_delivery_data import LinkMsg
from qep_delivery_data import TextMsg
from qep_delivery_data import send_custom_msg_to_group


def sample():
    pic = open("img.png", "rb")
    pic_base64 = base64.b64encode(pic.read()).decode('utf-8')
    pic.close()

    msg = InfoFlowMsgContent(
        self.group_id,
        [
            AtUserMsg(["guyouda"]),
            TextMsg("测试发送自定义消息\n"),
            LinkMsg("https://testone.baidu-int.com/\n"),
            ImageMsg(pic_base64),
        ]
    )
    result = send_custom_msg_to_group(self.app_key, self.app_secret, msg)
```
