# 2、手动输入知识库链接采集提交
依赖版本升级

```
qep-delivery-data==0.2.8
qep-delivery-component==0.2.18
qep-delivery-tools==0.4.16
```
Agent下发一个表单组件，类型为 知识库抓取类型

```
from qep_delivery_component import ComponentOperateConfig
from qep_delivery_component import FormComponent
from qep_delivery_component import KuCrawlFormDetail
from qep_delivery_tools import BaseTool
from qep_delivery_tools import Parameter
from qep_delivery_tools import ToolResult

...

class FormTool(BaseTool):
    """
    表单组件
    测试表单提交能力
    """

    def __init__(self):
        super().__init__("xxx")

    def execute(self, parameter: Parameter) -> ToolResult:
        component_data = [
            KuCrawlFormDetail(
                name="documents",
                display_value="知识库链接",
                required=True,
                max_length=8,
                default_value=[],
                tip="请输入知识库文档链接"
            )
        ]
        
        # 封装为一个列表组件
        return ToolResult(
            self.tool_id,
            FormComponent(
                component_operate_config=ComponentOperateConfig(), 
                component_data=component_data,
                strategy_id=30010  // 替换成你的麒麟策略ID，提交后会把用户填写的内容调起这个接口
            ) 
        )

```
