折线图组件

qep-delivery-component >= 0.4.0

# 示例代码
```
from qep_delivery_component import ComponentOperateConfig
from qep_delivery_component import LineChart
from qep_delivery_component import LineChartComponent
from qep_delivery_component import LineInfo
from qep_delivery_tools import Parameter
from qep_delivery_tools import ToolResult
from qep_delivery_tools import BaseTool


class XxxTool(BaseTool):
    """
    Xxx工具
    """

    def __init__(self):
        super().__init__(自己注册的ToolId)


    def execute(self, parameter: Parameter) -> ToolResult:
        """
        展示组件结果封装
        """
        line_list = [
            LineInfo(name='A', data=[10, 20, 30, 40, 50, 60, 70]),
            LineInfo(name='B', data=[15, 25, 35, 45, 55, 65, 75]),
            LineInfo(name='C', data=[30, 20, 10, 60, 90, 81, 100], color="#91CC75")
        ]
        line_chart = LineChart(
            x_axis_data=["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"],
            series_data=line_list
        )
        component_info = LineChartComponent(component_operate_config=None, component_data=line_chart)
        return ToolResult(self.tool_id, component_info)
```
