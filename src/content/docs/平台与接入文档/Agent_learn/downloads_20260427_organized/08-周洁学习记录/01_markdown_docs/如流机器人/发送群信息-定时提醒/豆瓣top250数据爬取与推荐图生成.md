import os
import json
import time
import requests
import lxml.etree as etree
from openpyxl import Workbook


def process_person_info(person_info):
    """
    处理人员信息
    """
    person_list = []
    for item in person_info:
        person = item['data'][0]['name']
        person_list.append(person)
    person_str = '/'.join(person_list)
    if len(person_str) > 20:
        person_str = person_str[:19] + '…'
    return person_str


def get_movie_info(doubanId):
    """
    获取电影详情
    """
    url = 'https://movie.querydata.org/api?id=%s' % doubanId
    req = requests.get(url)
    req_res = json.loads(req.content)
    base_info = req_res['data'][0]
    movie_info = {
        'doubanId': doubanId,
        'name': base_info['name'],
        'name_org': req_res['originalName'],
        'poster': base_info['poster'],
        'director': process_person_info(req_res['director']),
        'actor': process_person_info(req_res['actor']),
        'time': req_res['dateReleased'],
        'country': base_info['country'],
        'type': base_info['genre'],
        'duration': '%s分钟' % int(req_res['duration'] / 60),
        'description': base_info['description'],
        'score_db': req_res['doubanRating'] if req_res['doubanRating'] != 'null' else '暂无',
        'vote_db': req_res['doubanVotes'],
        'score_imdb': req_res['imdbRating'] if req_res['imdbRating'] != 'null' else '暂无',
        'vote_imdb': req_res['imdbVotes'],
        'score_rot': '暂无',
        'vote_rot': 0,
    }
    if 'rottenRating' in req_res and req_res['rottenRating'] != 'null':
        req_res['score_rot'] = req_res['rottenRating']
        req_res['vote_rot'] = req_res['rottenVotes']

    print(json.dumps(movie_info, indent=4, ensure_ascii=False))

    poster_url = 'https://imageserver.querydata.org/api?url=%s&width=400&format=jpg' % movie_info['poster']
    r = requests.get(poster_url)
    tmp_path = "poster/%s.jpg" % doubanId
    with open(tmp_path, "wb") as f:
        f.write(r.content)

    return movie_info


def get_source(top250, page):
    """
    爬取豆瓣电影id
    """
    url = 'https://movie.douban.com/top250?start={}'.format(page)
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) '
                      'Chrome/73.0.3683.103 Safari/537.36'}
    res = requests.get(url=url, headers=headers)
    html = etree.HTML(res.text)
    for i in range(1, 26):
        doubanId_cont = html.xpath("/html/body/div[3]/div[1]/div/div[1]/ol/li[{}]/div/div[2]/div[1]/a/@href".format(i))
        name_cont = html.xpath("/html/body/div[3]/div[1]/div/div[1]/ol/li[{}]/div/div[2]/div[1]/a/span[1]/text()".format(i))
        doubanId = doubanId_cont[0].split('/')[-2]
        name = name_cont[0]
        top250[doubanId] = name


def get_douban_data(patch):
    """
    patch: 1~5 或 None
    None表示top1～250，1表示top1～50， 2表示top51～100，依次类推
    """
    top250 = {}
    if patch is None:
        for i in range(0, 250, 25):
            get_source(top250, i)
        with open('top250_id.json', "w", encoding='utf-8') as f:
            json.dump(top250, f, ensure_ascii=False, indent=4, )
    else:
        get_source(top250, 50 * patch)
        get_source(top250, 50 * patch + 24)

    print(json.dumps(top250, ensure_ascii=False, indent=4))

    return list(top250.keys())


def run(patch=None):
    """
    爬取电影信息
    """
    res = []
    doubanId_list = get_douban_data(patch)
    for doubanId in doubanId_list:
        try:
            movie_info = get_movie_info(doubanId)
            res.append(movie_info)
        except:
            print('run error: %s' % doubanId)
        time.sleep(30)

    name = 'all_%s' % patch if patch is not None else 'all'
    with open(name, "w", encoding='utf-8') as f:
        json.dump(res, f, ensure_ascii=False, indent=4, )


def save_excel(data, title):
    """
    写入Excel表格
    data = [[第一行内容],[第二行内容]]
    """
    if data is None or len(data) == 0:
        return
    wb = Workbook()
    ws1 = wb.active
    ws1.title = title
    for row in data:
        ws1.append(row)
    wb.save('data/%s.xlsx' % title)


def save_movie_info_all():
    """
    top250信息汇总保存
    """
    with open('data/movie_conf.json', "rb") as f:
        movie_conf_info = json.load(f)
    with open('data/api.json', "rb") as f2:
        top250 = json.load(f2)

    key_list = ['doubanId', 'name', 'name_org', 'poster', 'director', 'actor', 'time',
                'country', 'type', 'duration', 'description', 'score_db', 'vote_db',
                'score_imdb', 'vote_imdb', 'score_rot', 'vote_rot']
    null_movie = {}
    for i in key_list:
        null_movie[i] = None

    movie_info = {}
    for item in top250:
        movie_info[item['doubanId']] = item

    data_all = []
    ranker = 1
    for order in movie_conf_info:
        movie_tmp = {'rank': ranker, 'order': order}
        doubanId = movie_conf_info[order]['doubanId']
        if doubanId in movie_info:
            movie_tmp.update(movie_info[doubanId])
        else:
            print(doubanId)
            movie_tmp.update(null_movie)
            movie_tmp['doubanId'] = doubanId
        if os.path.exists('poster/%s.jpg' % doubanId):
            movie_tmp['poster'] = 'poster/%s.jpg' % doubanId
        else:
            movie_tmp['poster'] = None
        data_all.append(movie_tmp)
        ranker += 1

    with open('data/top250.json', "w", encoding='utf-8') as f:
        json.dump(data_all, f, ensure_ascii=False, indent=4, )

    excel_data = []
    title = list(data_all[0].keys())
    excel_data.append(title)
    for item in data_all:
        tmp = []
        for key in data_all[0].keys():
            tmp.append(item[key])
        excel_data.append(tmp)

    save_excel(excel_data, 'top250')


if __name__ == "__main__":
    run()

```
def hsv_color_diff(color1, color2):
    """
    计算色差
    """
    H_1, S_1, V_1 = colorsys.rgb_to_hsv(color1[0], color1[1], color1[2])
    H_2, S_2, V_2 = colorsys.rgb_to_hsv(color2[0], color2[1], color2[2])
    R, angle_ = 100, 30
    h = R * math.cos(angle_ / 180 * math.pi)
    r = R * math.sin(angle_ / 180 * math.pi)
    x1 = r * V_1 * S_1 * math.cos(H_1 / 180 * math.pi)
    y1 = r * V_1 * S_1 * math.sin(H_1 / 180 * math.pi)
    z1, z2 = h * (1 - V_1), h * (1 - V_2)
    x2 = r * V_2 * S_2 * math.cos(H_2 / 180 * math.pi)
    y2 = r * V_2 * S_2 * math.sin(H_2 / 180 * math.pi)
    dx = x1 - x2
    dy = y1 - y2
    dz = z1 - z2
    return int(math.sqrt(dx * dx + dy * dy + dz * dz))

```
import copy
import json
import math
import traceback
import colorsys

import cv2
from pptx import Presentation
from pptx.dml.color import RGBColor
from haishoku.haishoku import Haishoku


def get_movie_info(doubanId):
    """
    从配置中读取电影信息
    """
    with open('data/top250.json', "rb") as f:
        movie_info_list = json.load(f)
    for movie_info in movie_info_list:
        if movie_info['doubanId'] == str(doubanId):
            print(json.dumps(movie_info, indent=4, ensure_ascii=False))
            return movie_info


def hsv_color_diff(color1, color2):
    """
    计算色差
    """
    H_1, S_1, V_1 = colorsys.rgb_to_hsv(color1[0], color1[1], color1[2])
    H_2, S_2, V_2 = colorsys.rgb_to_hsv(color2[0], color2[1], color2[2])
    R, angle_ = 100, 30
    h = R * math.cos(angle_ / 180 * math.pi)
    r = R * math.sin(angle_ / 180 * math.pi)
    x1 = r * V_1 * S_1 * math.cos(H_1 / 180 * math.pi)
    y1 = r * V_1 * S_1 * math.sin(H_1 / 180 * math.pi)
    z1, z2 = h * (1 - V_1), h * (1 - V_2)
    x2 = r * V_2 * S_2 * math.cos(H_2 / 180 * math.pi)
    y2 = r * V_2 * S_2 * math.sin(H_2 / 180 * math.pi)
    dx = x1 - x2
    dy = y1 - y2
    dz = z1 - z2
    return int(math.sqrt(dx * dx + dy * dy + dz * dz))


def get_core_color(doubanId):
    """
    海报主色调
    """
    haishoku = Haishoku.loadHaishoku('poster/%s.jpg' % doubanId)
    color_group = haishoku.palette
    core_color = None
    core_diff_max = 0
    for item in color_group:
        color = item[1]
        color_diff = hsv_color_diff(color, (255, 255, 255))
        if color_diff > core_diff_max:
            core_color = color
        if color_diff > 15000:
            core_color = color
            break
    return core_color


def process_pptx_slide(slide, movie_info):
    """
    生成单页ppt内容
    """
    shapes = slide.shapes

    core_color = get_core_color(movie_info['doubanId'])
    fill = shapes[0].fill
    fill.fore_color.rgb = RGBColor(core_color[0], core_color[1], core_color[2])

    change_map = {
        6: 'director', 7: 'actor', 8: 'time', 9: 'type', 10: 'duration',
    }
    for key in change_map:
        shapes[key].text_frame.paragraphs[0].runs[1].text = str(movie_info[change_map[key]])
    shapes[8].text_frame.paragraphs[0].runs[1].text = '%s/%s' % (movie_info['time'], movie_info['country'])
    change_map2 = {
        5: 'name', 11: 'description', 19: 'score_db', 17: 'score_imdb', 15: 'score_rot',
        20: 'vote_db', 18: 'vote_imdb', 16: 'vote_rot',
    }
    for key in change_map2:
        shapes[key].text_frame.paragraphs[0].runs[0].text = str(movie_info[change_map2[key]])

    pos = 21
    e = shapes[pos].element
    slide.shapes.add_picture('poster/%s.jpg' % movie_info['doubanId'], shapes[pos].left, shapes[pos].top,
                             shapes[pos].width, shapes[pos].height)
    e.getparent().remove(e)


def creat_img_pptx_single(doubanId):
    """
    生成中间文件pptx
    """
    movie_info = get_movie_info(doubanId)

    ppt = Presentation('data/tmp.pptx')
    slide = ppt.slides[0]  # 定位到ppt的第一张幻灯片
    shapes = slide.shapes  # 获取所有的shape
    for index, shape in enumerate(shapes):
        if shape.has_text_frame:
            text_frame = shape.text_frame
            print('%s:%s' % (index, text_frame.text))

    process_pptx_slide(slide, movie_info)

    ppt.save('data/ppt/%s.pptx' % movie_info['doubanId'])


def duplicate_slide(pres, index):
    """
    复制幻灯片
    """
    template = pres.slides[index]
    blank_slide_layout = pres.slide_layouts[6]
    copied_slide = pres.slides.add_slide(blank_slide_layout)

    for shp in template.shapes:
        if 'Picture' in shp.name or '图片' in shp.name:
            icon_id = int(shp.name.split(' ')[-1])
            icon_map = {8: 'douban', 10: 'imdb', 11: 'rot'}
            copied_slide.shapes.add_picture('data/icon/%s.png' % icon_map[icon_id],
                                            shp.left, shp.top, shp.width, shp.height)
        else:
            el = shp.element
            newel = copy.deepcopy(el)
            copied_slide.shapes._spTree.insert_element_before(newel, 'p:extLst')
    return copied_slide


def creat_img_pptx_all():
    """
    批量生成中间文件pptx
    """
    with open('data/top250.json', "rb") as f:
        movie_info_list = json.load(f)

    ppt = Presentation('data/tmp.pptx')

    for movie_info in movie_info_list:
        slide = duplicate_slide(ppt, 0)
        try:
            process_pptx_slide(slide, movie_info)
        except:
            print('error: %s, mgs: %s' % (movie_info, traceback.format_exc()))

    ppt.save('data/ppt/all.pptx')


def save_movie_card():
    """
    裁剪卡片并保存
    """
    with open('data/top250.json', "rb") as f:
        movie_info_list = json.load(f)

    for index, movie_info in enumerate(movie_info_list):
        img = cv2.imread('data/ppt/all/幻灯片%s.png' % str(index + 2))
        cropped_image = img[0:810, 0:1820]
        card = cv2.resize(cropped_image, (905, 400))
        cv2.imwrite("data/card/%s.png" % movie_info['doubanId'], card)


if __name__ == "__main__":
    # creat_img_pptx_all()
    save_movie_card()
