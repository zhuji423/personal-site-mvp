package com.baidu.qamate.core.apismartmgt.utils;

import com.alibaba.fastjson2.JSON;
import com.alibaba.fastjson2.JSONObject;
import lombok.Data;
import lombok.extern.slf4j.Slf4j;

@Slf4j
public class AlertConfigUtils {

      @Data
      public static class AlertConfig {
          private Boolean enabled;
          private Integer failThreshold;
          private Integer timeoutThreshold;
          private String channels;
          private String callbackUrl;
          private Boolean isKirin;
      }

      /**
       * 从plan_params解析报警配置
       */
      public static AlertConfig parseAlertConfig(String planParams) {
          try {
              if (planParams == null || planParams.trim().isEmpty()) {
                  return null;
              }

              JSONObject json = JSON.parseObject(planParams);
              JSONObject alarmInfo = json.getJSONObject("alarmInfo");

              if (alarmInfo == null) {
                  return null;
              }

              AlertConfig config = new AlertConfig();
              config.setEnabled(alarmInfo.getBooleanValue("enabled", false));
              config.setFailThreshold(alarmInfo.getIntValue("failThreshold", 3));
              config.setTimeoutThreshold(alarmInfo.getIntValue("timeoutThreshold", 300));
              config.setCallbackUrl(alarmInfo.getString("callbackUrl"));
              config.setIsKirin(alarmInfo.getBooleanValue("isKirin", false));

              JSONObject channels = alarmInfo.getJSONObject("channels");
              if (channels != null) {
                  config.setChannels(channels.toJSONString());
              }

              return config;
          } catch (Exception e) {
              log.error("解析报警配置失败: {}", planParams, e);
              return null;
          }
      }
}

  private boolean shouldTriggerAlert(ApiPlanExeEntity exe, AlertConfig config) {
  // 条件1：状态是失败，并且连续失败次数达到阈值
  if ("1".equals(exe.getStatus())) { // 1代表失败
  int failCount = getContinuousFailCount(exe.getPlanId(), config.getFailThreshold());
  if (failCount >= config.getFailThreshold()) {
  log.info("【报警】触发原因：连续失败达到阈值, failCount={}, threshold={}",
  failCount, config.getFailThreshold());
  return true;
  }
  }

  // 条件2：执行超时
  if (exe.getEndTime() != null && exe.getStartTime() != null) {
  long durationSeconds = exe.getEndTime() - exe.getStartTime();
  if (durationSeconds > config.getTimeoutThreshold()) {
  log.info("【报警】触发原因：执行超时, duration={}秒, timeout={}秒",
  durationSeconds, config.getTimeoutThreshold());
  return true;
  }
  }

  return false;
  }

@Override
public void planExeEnd(DispatchExeRequestVo dispatchExeRequestVo) {
// ===== 保留原有逻辑 =====
log.info("planExeEnd: {}", dispatchExeRequestVo);

      var plan = apiPlanMapper.selectById(dispatchExeRequestVo.getPlanId());
      if (plan == null || plan.getIsDel()) {
          throw new CoreException(StatusEnum.PLAN_NOT_EXIST);
      }

      if (StringUtils.isNotEmpty(dispatchExeRequestVo.getExeId())) {
          var planExe = apiPlanExeMapper.selectById(dispatchExeRequestVo.getExeId());
          if (planExe == null) {
              throw new CoreException(StatusEnum.EXECUTION_NOT_EXIST);
          }

          // ... 所有原有逻辑保持不变 ...
      }

      // ===== 报警逻辑（核心！）=====
      try {
          log.info("【报警】执行完成，开始报警检查, planId: {}, exeId: {}",
              dispatchExeRequestVo.getPlanId(),
              dispatchExeRequestVo.getExeId());

          // 1. 查询测试计划配置
          ApiPlanEntity plan = apiPlanMapper.selectById(dispatchExeRequestVo.getPlanId());
          if (plan == null) {
              log.warn("【报警】测试计划不存在, planId: {}", dispatchExeRequestVo.getPlanId());
              return;
          }

          // 2. 获取执行记录
          ApiPlanExeEntity exe = apiPlanExeMapper.selectById(dispatchExeRequestVo.getExeId());
          if (exe == null) {
              log.warn("【报警】执行记录不存在, exeId: {}", dispatchExeRequestVo.getExeId());
              return;
          }

          // 3. 解析报警配置
          AlertConfig config = AlertConfigUtils.parseAlertConfig(plan.getPlanParams());
          if (config == null || !config.getEnabled()) {
              log.debug("【报警】未配置报警或不启用, planId: {}", plan.getId());
              return;
          }

          // 4. 检查是否满足报警条件
          if (!shouldTriggerAlert(exe, config)) {
              log.info("【报警】不满足报警条件, planId: {}, exeId: {}",
                  plan.getId(), exe.getId());
              return;
          }

          // 5. 生成报警记录
          AlertHistoryEntity history = buildAlertHistory(plan, exe, config);

          // 6. 记录到数据库
          boolean recordSuccess = alertHistoryService.recordAlertHistory(history);
          if (!recordSuccess) {
              log.error("【报警】报警记录失败, alertId: {}", history.getAlertId());
              return;
          }
