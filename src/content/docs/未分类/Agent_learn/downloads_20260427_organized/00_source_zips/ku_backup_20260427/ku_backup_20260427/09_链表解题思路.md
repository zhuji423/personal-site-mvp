## 双指针
### 合并两个有序链表
#### [易]合并两个有序链表
[https://leetcode.cn/problems/merge-two-sorted-lists/](https://leetcode.cn/problems/merge-two-sorted-lists/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def mergeTwoLists(self, list1, list2):
        """
        :type list1: Optional[ListNode]
        :type list2: Optional[ListNode]
        :rtype: Optional[ListNode]
        """
        dummy = ListNode(0)
        p = dummy
        p1, p2 = list1, list2
        while p1 and p2:
            if p1.val < p2.val:
                p.next = ListNode(p1.val)
                p1 = p1.next
            else:
                p.next = ListNode(p2.val)
                p2 = p2.next
            p = p.next
        p.next = p1 if p1 else p2
        return dummy.next
```
### 单链表的分解
#### [中]分隔链表
[https://leetcode.cn/problems/partition-list/description/](https://leetcode.cn/problems/partition-list/description/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def partition(self, head, x):
        """
        :type head: Optional[ListNode]
        :type x: int
        :rtype: Optional[ListNode]
        """
        dummy1 = ListNode(0)
        dummy2 = ListNode(0)
        p1, p2 = dummy1, dummy2
        p = head
        while p:
            if p.val < x:
                p1.next = ListNode(p.val)
                p1 = p1.next
            else:
                p2.next = ListNode(p.val)
                p2 = p2.next
            p = p.next
        p1.next = dummy2.next
        return dummy1.next
```
### 合并 k 个有序链表
#### [难]合并 K 个升序链表
[https://leetcode.cn/problems/merge-k-sorted-lists/](https://leetcode.cn/problems/merge-k-sorted-lists/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def mergeKLists(self, lists):
        """
        :type lists: List[Optional[ListNode]]
        :rtype: Optional[ListNode]
        """
        dummy = ListNode(0)
        p = dummy
        data = []
        for node in lists:
            if node:
                heapq.heappush(data, (node.val, node))
        while len(data):
            _, cur = heapq.heappop(data)
            p.next = ListNode(cur.val)
            p = p.next
            if cur.next:
                heapq.heappush(data, (cur.next.val, cur.next))

        return dummy.next
```

# 返回链表的倒数第 k 个节点
def findFromEnd(head: ListNode, k: int) -> ListNode:
    l, r = head, head
    for _ in range(k):
        r = r.next
    while r:
        l = l.next
        r = r.next
    return l

#### [中]删除链表的倒数第 N 个结点
[https://leetcode.cn/problems/remove-nth-node-from-end-of-list/](https://leetcode.cn/problems/remove-nth-node-from-end-of-list/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def removeNthFromEnd(self, head, n):
        """
        :type head: Optional[ListNode]
        :type n: int
        :rtype: Optional[ListNode]
        """
        dummy = ListNode(0, head)
        l, r = dummy, dummy
        # 找到倒数n+1节点
        for _ in range(n + 1):
            r = r.next
        while r:
            l = l.next
            r = r.next
        # 删除l的下一个
        l.next = l.next.next
        return dummy.next
        
```

### 判断链表是否包含环
#### [易]环形链表
[https://leetcode.cn/problems/linked-list-cycle/description/](https://leetcode.cn/problems/linked-list-cycle/description/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution(object):
    def hasCycle(self, head):
        """
        :type head: ListNode
        :rtype: bool
        """
        l, r = head, head
        while r and r.next:
            l = l.next
            r = r.next.next
            if l == r:
                return True 
        return False
        
```

# Definition for singly-linked list.
# class ListNode(object): 
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution(object):
    def detectCycle(self, head):
        """
        :type head: ListNode
        :rtype: ListNode
        """
        l, r = head, head
        while r and r.next:
            l = l.next
            r = r.next.next
            if l == r:
                break
        if not r or not r.next:
            return None
        
        l = head
        while l != r:
            l = l.next
            r = r.next
        return l
```
### 两个链表是否相交
#### [易]相交链表
[https://leetcode.cn/problems/intersection-of-two-linked-lists/](https://leetcode.cn/problems/intersection-of-two-linked-lists/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution(object):
    def getIntersectionNode(self, headA, headB):
        """
        :type head1, head1: ListNode
        :rtype: ListNode
        """
        l, r = headA, headB
        while l != r:
            l = l.next if l else headB
            r = r.next if r else headA
        return l
```
## 单链表的花式反转
#### [易]反转链表
[https://leetcode.cn/problems/reverse-linked-list/description/](https://leetcode.cn/problems/reverse-linked-list/description/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def reverseList(self, head):
        """
        :type head: Optional[ListNode]
        :rtype: Optional[ListNode]
        """
        p, pre = head, None
        while p:
            tmp = p.next
            p.next = pre
            pre = p
            p = tmp
        return pre
```

# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def reverseBetween(self, head, left, right):
        """
        :type head: Optional[ListNode]
        :type left: int
        :type right: int
        :rtype: Optional[ListNode]
        """
        # pre_l: left 前一个节点
        # l: 第 left 个节点
        # r: 第 right 个节点
        # next_r: right 后一个节点

        # 寻找pre_l
        dummy = ListNode(0, head)
        pre_l = dummy
        for _ in range(left - 1):
            pre_l = pre_l.next
        l = pre_l.next

        # 寻找next_r，并切断right
        r = l
        for _ in range(right - left):
            r = r.next
        next_r = r.next
        r.next = None

        # 反转left
        p, pre = l, None
        while p:
            tmp = p.next
            p.next = pre
            pre = p
            p = tmp
        
        # 拼接
        l.next = next_r
        pre_l.next = pre
        
        return dummy.next
```
#### [中]旋转链表
[https://leetcode.cn/problems/rotate-list/description/](https://leetcode.cn/problems/rotate-list/description/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = None

class Solution(object):
    def rotateRight(self, head, k):
        """
        :type head: ListNode
        :type k: int
        :rtype: ListNode
        """
        if not head or not head.next or k == 0:
            return head

        def len(p):
            res = 0
            while p:
                res += 1
                p = p.next
            return res
        
        k = k % len(head)
        if k == 0:
            return head

        # 倒数第k+1个节点
        l, r = head, head
        for _ in range(k + 1):
            r = r.next
        while r:
            l = l.next
            r = r.next

        node = l.next
        l.next = None
        p = node
        while p.next:
            p = p.next
        p.next = head

        return node
```
## 回文链表
#### [易]回文链表
空间复杂度O(1)完成就不简单了

[https://leetcode.cn/problems/palindrome-linked-list/](https://leetcode.cn/problems/palindrome-linked-list/)

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def isPalindrome(self, head):
        """
        :type head: Optional[ListNode]
        :rtype: bool
        """
        if not head or not head.next:
            return True

        # 找中点
        l, r = head, head
        while r and r.next:
            l = l.next
            r = r.next.next
        
        # 反转后半部分链表
        p, pre = l, None
        while p:
            tmp = p.next
            p.next = pre
            pre = p
            p = tmp

        # 比较前后两部分
        l, r = head, pre
        while r:
            if l.val != r.val:
                return False
            l = l.next
            r = r.next
        return True

```
