class Solution(object):
    def sortArray(self, nums):
        """
        :type nums: List[int]
        :rtype: List[int]
        """
        def part(nums, l, r):
            val = nums[r]
            i = l
            for j in range(l, r):
                if nums[j] < val:
                    nums[i], nums[j] = nums[j], nums[i]
                    i += 1
            nums[i], nums[r] = nums[r], nums[i]
            return i

        def sort(nums, l, r):
            if l < r:
                index = part(nums, l, r)
                sort(nums, l, index - 1)
                sort(nums, index + 1, r)

        sort(nums, 0, len(nums) - 1)
        return nums
```

class Solution(object):

        """
        :type nums: List[int]
        :rtype: List[int]
        """
        if len(nums) <= 1: 
            return nums
        
        mid = len(nums) // 2

        left = self.sortArray(nums[:mid])
        right = self.sortArray(nums[mid:])

        res = []
        i, j = 0, 0
        while i < len(left) and j < len(right):
            if left[i] < right[j]:
                res.append(left[i])
                i += 1
            else:
                res.append(right[j])
                j += 1
        res.extend(left[i:])
        res.extend(right[j:])

        return res

堆排序：

```
class Solution(object):
    def sortArray(self, nums):
        """
        :type nums: List[int]
        :rtype: List[int]
        """
        def sift_down(nums, i, size):
            """下沉操作，将堆顶元素下沉到正确的位置"""
            largest = i
            left = 2 * i + 1  # 二叉堆是完全二叉树
            right = 2 * i + 2

            if left < size and nums[largest] < nums[left]:
                largest = left
            if right < size and nums[largest] < nums[right]:
                largest = right
            if largest != i:
                nums[largest], nums[i] = nums[i], nums[largest]
                sift_down(nums, largest, size)
            
        def heapify(nums):
            for i in range(len(nums) // 2, -1, -1):
                sift_down(nums, i, len(nums))

        # 构建最大堆
        heapify(nums)

        # 堆排序
        for i in range(len(nums) - 1, 0, -1):
            nums[0], nums[i] = nums[i], nums[0]
            # 堆化剩下部分
            sift_down(nums, 0, i)
        return nums
```
#### [中]数组中的第K个最大元素
[https://leetcode.cn/problems/kth-largest-element-in-an-array/description/](https://leetcode.cn/problems/kth-largest-element-in-an-array/description/)

```
class Solution(object):
    def findKthLargest(self, nums, k):
        """
        :type nums: List[int]
        :type k: int
        :rtype: int
        """
        # 大小为 K 的最小堆，每次插入一个新元素时都进行 heappushpop，这样堆顶总是保留当前的第 K 大元素
        heap = nums[:k]
        heapq.heapify(heap)

        for i in nums[k:]:
            if i < heap[0]:
                continue
            heapq.heappushpop(heap, i)
        return heap[0]
```

### 快慢指针
#### [易]删除有序数组中的重复项
****[https://leetcode.cn/problems/remove-duplicates-from-sorted-array/](https://leetcode.cn/problems/remove-duplicates-from-sorted-array/)

慢指针 `slow` 走在后面，快指针 `fast` 走在前面探路，找到一个不重复的元素就赋值给 `slow` 并让 `slow` 前进一步

```
class Solution(object):
    def removeDuplicates(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        if not nums:
            return 0
            
        l, r = 0, 1
        while r < len(nums):
            if nums[l] != nums[r]:
                l += 1
                nums[l] = nums[r]
            r += 1
        return l + 1
```

#### [中]最长回文子串
[https://leetcode.cn/problems/longest-palindromic-substring/description/](https://leetcode.cn/problems/longest-palindromic-substring/description/)

找回文串的难点在于，回文串的的长度可能是奇数也可能是偶数，解决该问题的核心是**从中心向两端扩散的双指针技巧**。

```
class Solution(object):
    def longestPalindrome(self, s):
        """
        :type s: str
        :rtype: str
        """
        def sub(l, r, s):
            while l >= 0 and r < len(s) and s[l] == s[r]:
                l -=1
                r += 1 
            return s[l + 1: r]
        
        res = ''
        for i in range(len(s)):
            l1 = sub(i, i, s)
            l2 = sub(i, i + 1, s)
            res = res if len(l1) < len(res) else l1
            res = res if len(l2) < len(res) else l2
        return res
```
#### **[中]最短无序连续子数组**
* 将原数组 `nums` 拷贝一份并排序，记为 `temp`。
* 比较 `nums` 和 `temp` 中对应位置的元素，找到**第一个不同的左边界 **`left`，和**最后一个不同的右边界 **`right`

[https://leetcode.cn/problems/shortest-unsorted-continuous-subarray/](https://leetcode.cn/problems/shortest-unsorted-continuous-subarray/)

```
class Solution(object):
    def findUnsortedSubarray(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        tmp = sorted(nums)
        left, right = -1, len(nums)
        l, r = 0, len(nums) - 1
        while l < len(nums):
            if nums[l] != tmp[l]:
                left = l
                break
            l += 1
        while r >= 0:
            if nums[r] != tmp[r]:
                right = r
                break
            r -= 1
        if left == -1 and right == len(nums):
            return 0
        return r - l + 1
```

class Solution(object):

        """
        :type nums: List[int]
        :type target: int
        :rtype: List[List[int]]
        """
        res = []
        nums.sort()
        for i in range(len(nums) - 3):

                            r -= 1
                        l += 1
                        r -= 1

## 二分搜索
#### [中]在排序数组中查找元素的第一个和最后一个位置
[https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/](https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/)

```
class Solution(object):
    def searchRange(self, nums, target):
        """
        :type nums: List[int]
        :type target: int
        :rtype: List[int]
        """
        def find(nums, i):
            l, r = i, i
            while l > 0 and nums[l] == nums[l - 1]:
                l -= 1
            while r < len(nums) - 1 and nums[r] == nums[r + 1]:
                r += 1
            return l, r

        l, r = 0, len(nums) - 1
        while l <= r:
            mid = (l + r) // 2
            if nums[mid] < target:
                l = mid + 1
            elif nums[mid] > target:
                r = mid - 1
            else:
                return find(nums, mid)
        return [-1, -1]
```

class Solution(object):
    def reverseWords(self, s):
        """
        :type s: str
        :rtype: str
        """
        # 分割字符串并过滤空字符串
        words = [word for word in s.split(' ') if word != '']

```
class Solution(object):
    def reverseWords(self, s):
        """
        :type s: str
        :rtype: str
        """
        # 将字符串转换为列表以便原地修改
        s = list(s)

        def trim_spaces(s):
            while s and s[0] == ' ':
                s.pop(0)
            while s and s[-1] == ' ':
                s.pop(-1)
            i = 0
            while i < len(s):
                if s[i] == ' ' and i > 0 and s[i - 1] == ' ':
                    s.pop(i)
                else:
                    i += 1

        def reverse(s, l, r):
            while l <= r:
                s[l], s[r] = s[r], s[l]
                l += 1
                r -= 1

        def reverse_word(s):
            l, r = 0, 0
            while l < len(s):
                while r < len(s) and s[r] != ' ':
                    r += 1
                reverse(s, l, r - 1)
                l = r + 1
                r = l

        # 1. 去除多余空格（原地）
        trim_spaces(s)
        # 2. 反转整个字符串
        reverse(s, 0, len(s)-1)
        # 3. 逐个反转单词
        reverse_word(s)
        
        return ''.join(s)
```

class Solution(object):
    def spiralOrder(self, matrix):
        """
        :type matrix: List[List[int]]
        :rtype: List[int]
        """
        if not matrix:
            return []
        res = []
        m, n = len(matrix), len(matrix[0])
        left, right, up, down = 0, n - 1, 0, m - 1
        i, j = 0, 0
        path = [(0, 1), (1, 0), (0, -1), (-1, 0)]  # 0右  1下  2左  3上
        pos = 0  # 0右  1下  2左  3上
        while len(res) != m * n:
            res.append(matrix[i][j])
            if pos == 0 and j == right:
                pos += 1
                up += 1
            elif pos == 1 and i == down:
                pos += 1
                right -= 1
            elif pos == 2 and j == left:
                pos += 1
                down -= 1
            elif pos == 3 and i == up:
                pos += 1
                left += 1

            pos %= 4
            i += path[pos][0]
            j += path[pos][1]
            
        return res

class Solution(object):
    def rotate(self, matrix):
        """
        :type matrix: List[List[int]]
        :rtype: None Do not return anything, modify matrix in-place instead.
        """
        n = len(matrix)
        
        # 先转置矩阵（行列互换）
        for i in range(n):
            for j in range(i, n):
                matrix[i][j], matrix[j][i] = matrix[j][i], matrix[i][j]
        
        # 然后反转每一行
        for row in matrix:
            row.reverse()

[https://leetcode.cn/problems/longest-substring-without-repeating-characters/](https://leetcode.cn/problems/longest-substring-without-repeating-characters/)

```
class Solution(object):
    def lengthOfLongestSubstring(self, s):
        """
        :type s: str
        :rtype: int
        """
        l, r = 0, 0
        max_s = 0
        d = deque()
        while r < len(s):
            cur = s[r]
            while cur in d:
                d.popleft()
                l += 1
            max_s = max(max_s, r - l + 1)
            d.append(cur)
            r += 1
        return max_s
```
#### [中]找到字符串中所有字母异位词

class Solution(object):
    def findAnagrams(self, s, p):
        """
        :type s: str
        :type p: str
        :rtype: List[int]
        """
        if len(s) < len(p):
            return []

        s_dic, p_dic = defaultdict(int), defaultdict(int)
        for i in range(len(p)):
            s_dic[s[i]] += 1
            p_dic[p[i]] += 1

        res = []
        if s_dic == p_dic:
            res.append(0)
        
        l, r = 0, len(p) - 1
        while r < len(s) - 1:
            right = s[r + 1] # 要删的
            left = s[l] # 新加的
            if s_dic[left] == 1:
                del s_dic[left]
            else:
                s_dic[left] -= 1
            
            s_dic[right] += 1
            if s_dic == p_dic:
                res.append(l + 1)

            l += 1
            r += 1
        return res
```

class Solution(object):
    def checkInclusion(self, s1, s2):
        """
        :type s1: str
        :type s2: str
        :rtype: bool
        """
        if len(s1) > len(s2):
            return False

        s1_dic, s2_dic = defaultdict(int), defaultdict(int)
        for i in range(len(s1)):
            s1_dic[s1[i]] += 1
            s2_dic[s2[i]] += 1
        if s1_dic == s2_dic:
            return True

        l, r = 0, len(s1) - 1
        while r < len(s2) - 1:
            left = s2[l]
            right = s2[r + 1]
            if s2_dic[left] == 1:
                del s2_dic[left]
            else:
                s2_dic[left] -= 1
            s2_dic[right] += 1

            if s1_dic == s2_dic:
                return True
                
            l += 1
            r += 1
        return False
```
#### [难]滑动窗口最大值
[https://leetcode.cn/problems/sliding-window-maximum/](https://leetcode.cn/problems/sliding-window-maximum/)

最简单的写法，但超时了

```
class Solution(object):
    def maxSlidingWindow(self, nums, k):
        """
        :type nums: List[int]
        :type k: int
        :rtype: List[int]
        """
        wind = deque()
        res = []
        for i, c in enumerate(nums):
            if i < k - 1:
                wind.append(c)
                continue
            wind.append(c)
            max_val = max(wind)
            wind.popleft()
            res.append(max_val)
        return res

用单调队列：

```

class Solution(object):
    def maxSlidingWindow(self, nums, k):
        """
        :type nums: List[int]
        :type k: int
        :rtype: List[int]
        """
        maxq = deque()
        res = []
        for i, c in enumerate(nums):
            if i < k - 1:
                while maxq and maxq[-1] < c:
                    maxq.pop()
                maxq.append(c)
                continue

            # 入元素
            while maxq and maxq[-1] < c:
                maxq.pop()
            maxq.append(c)
            # 更新max
            max_val = maxq[0]
            res.append(max_val)
            # 出元素
            if maxq[0] == nums[i - k + 1]:

        """
        :type trips: List[List[int]]
        :type capacity: int
        :rtype: bool
        """
        N = 1000
        diff = [0] * (N + 2)
        for num, i, j in trips:
            diff[i] += num
            diff[j + 1] -= num
        
        res = [0] * N
        res[0] = diff[1]
        for i in range(1, N):
            res[i] = res[i - 1] + diff[i + 1]
        return all(i <= capacity for i in res)

```
class Solution(object):
    def maxSubarraySumCircular(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # 非循环部分 中间最大子数组和
        c_sum = max_sum = nums[0]
        for i in nums[1:]:
            c_sum = max(i, c_sum + i)
            max_sum = max(max_sum, c_sum)

        # 循环部分 总和 - 最小子数组和
        c_sum = min_sum = nums[0]
        for i in nums[1:]:
            c_sum = min(i, c_sum + i)
            min_sum = min(min_sum, c_sum)

        # 全是负的  循环部分：总和 - 最小子数组和 = 0
        if max_sum < 0:
            return max_sum

        return max(max_sum, sum(nums) - min_sum)

#### [中]优势洗牌
都先排序，遍历敌方从大到小，己方最大的能打就打，打不了就用最小的

[https://leetcode.cn/problems/advantage-shuffle/](https://leetcode.cn/problems/advantage-shuffle/)

class Solution(object):
    def advantageCount(self, nums1, nums2):
        """
        :type nums1: List[int]
        :type nums2: List[int]
        :rtype: List[int]
        """
        size = len(nums1)
        a_sort = sorted(nums1)
        b_sort = sorted([(value, i) for i, value in enumerate(nums2)], key=lambda x: x[0], reverse=True)

        res = [0] * size
        l, r = 0, size - 1
        for value, index in b_sort:
            if a_sort[r] > value:
                res[index] = a_sort[r]
                r -= 1
            else:
                res[index] = a_sort[l]
                l += 1
        return res
```

class Solution(object):
    def stoneGame(self, piles):
        """
        :type piles: List[int]
        :rtype: bool
        """
        # dp[i][j] 表示在 piles[i:j] 当前玩家（先手）与对手之间的最大分差
        n = len(piles)
        dp = [[0] * n for _ in range(n)]

        for i in range(n):
            dp[i][i] = piles[i]

        for length in range(2, n + 1):
            for i in range(n - length + 1):
                j = i + length - 1
                dp[i][j] = max(
                    piles[i] - dp[i + 1][j],  # 取左边
                    piles[j] - dp[i][j - 1]  # 取右边
                )

        return dp[0][n - 1] > 0  # 如果先手赢得分差 > 0，Alice 赢

class Solution(object):
    def multiply(self, num1, num2):
        """
        :type num1: str
        :type num2: str
        :rtype: str
        """
        if num1 == "0" or num2 == "0":
            return "0"

        m, n = len(num1), len(num2)
        res = [0] * (m + n)

        for i in range(m - 1, -1, -1):
            for j in range(n - 1, -1, -1):
                mul = int(num1[i]) * int(num2[j])
                p1, p2 = i + j, i + j + 1
                total = mul + res[p2]

                res[p2] = total % 10
                res[p1] += total // 10

        res = [str(i) for i in res]
        res = ''.join(res).lstrip('0')
        return res or "0"

class Solution(object):
    def countPrimes(self, n):
        """
        :type n: int
        :rtype: int
        """
        # 遍历从 2 开始，到 √n 为止的数字（大于 √n 都是非质数，0和1均为非质数）
        # 如果 i 是质数，那么把 i*i, i*i+i, i*i+2i, ... 均为非质数
        if n < 2:
            return 0

        is_prime = [True] * n
        is_prime[0:2] = [False, False]

        for i in range(2, int(n ** 0.5) + 1):
            if is_prime[i]:
                for j in range(i*i, n, i):
                    is_prime[j] = False

        return sum(is_prime)
```
#### [中]超级次方

```
class Solution(object):
    def superPow(self, a, b):
        """
        :type a: int
        :type b: List[int]
        :rtype: int
        """
        MOD = 1337

        # 快速幂函数
        def pow(x, n):
            x %= MOD
            result = 1
            while n:
                if n % 2:
                    result = (result * x) % MOD
                x = (x * x) % MOD
                n //= 2
            return result

        if not b:
            return 1

        last_digit = b.pop()
        part1 = self.superPow(a, b)
        part1 = pow(part1, 10)  # 前面的部分的10次方
        part2 = pow(a, last_digit)  # 当前最后一位贡献

        return (part1 * part2) % MOD
```
