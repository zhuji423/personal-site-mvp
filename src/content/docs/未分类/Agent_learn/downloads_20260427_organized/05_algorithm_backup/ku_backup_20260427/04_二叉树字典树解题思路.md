#### [中]最大二叉树
[https://leetcode.cn/problems/maximum-binary-tree/](https://leetcode.cn/problems/maximum-binary-tree/)

```
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution(object):
    def constructMaximumBinaryTree(self, nums):
        """
        :type nums: List[int]
        :rtype: Optional[TreeNode]
        """
        if not nums:
            return None

        max_val = max(nums)
        index = nums.index(max_val)

        root = TreeNode(max_val)
        root.left = self.constructMaximumBinaryTree(nums[:index])
        root.right = self.constructMaximumBinaryTree(nums[index+1:])
        return root
        
```

```
        root.right = self.constructFromPrePost(preorder[index+1:], postorder[index:-1])
        return root
```

## 后序便利
#### [中]寻找重复的子树
[https://leetcode.cn/problems/find-duplicate-subtrees/](https://leetcode.cn/problems/find-duplicate-subtrees/)

```
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution(object):
    def findDuplicateSubtrees(self, root):
        """
        :type root: Optional[TreeNode]
        :rtype: List[Optional[TreeNode]]
        """
        # 后序遍历能知道全貌
        res = []
        info = defaultdict(int)

        def dfs(root):
            if not root:
                return '#'
            val = '%s,%s,%s' % (dfs(root.left), dfs(root.right), root.val)
            info[val] += 1
            if info[val] == 2:
                res.append(root)
            return val
        dfs(root)
        return res
```

```
        :type root: Optional[TreeNode]
        :rtype: bool

        def helper(root, l, r):
            if not root:
                return True
```

```
        :type root: Optional[TreeNode]
        :type k: int
        :rtype: int

        self.k = k
        def dfs(root):
            if not root:
                return

            if self.k == 0:

        dfs(root)
        return self.res
```

```
        :rtype: Optional[TreeNode]

        self.res = 0

            if not root:

            root.val = self.res
            dfs(root.left)
        dfs(root) 
        return root
```

#### [中]删除二叉搜索树中的节点
[https://leetcode.cn/problems/delete-node-in-a-bst/](https://leetcode.cn/problems/delete-node-in-a-bst/)

```
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution(object):
    def deleteNode(self, root, key):
        """
        :type root: Optional[TreeNode]
        :type key: int
        :rtype: Optional[TreeNode]
        """
        def find_min(root):
            while root.left:
                root = root.left
            return root

        def del_node(root, key):
            if not root:
                return 
            if root.val < key:
                root.right = del_node(root.right, key)
            elif root.val > key:
                root.left = del_node(root.left, key)
            else:
                if not root.left:
                    return root.right
                if not root.right:
                    return root.left
                node = find_min(root.right)
                root.val = node.val
                root.right = del_node(root.right, node.val)
            return root
        
        return del_node(root, key)
        
```

```
class Solution(object):
    def numTrees(self, n):
        """
        :type n: int
        :rtype: int
        """
        # G(n) = G(0)*G(n-1) + G(1)*G(n-2) + ... + G(n-1)*G(0)
        dp = [0] * (n + 1)
        dp[0] = 1
        for i in range(1, n + 1):
            for j in range(i):
                dp[i] += dp[j] * dp[i - 1 - j]
        return dp[n]
```

```
        :type n: int
        :rtype: List[Optional[TreeNode]]
        """
        def build(l, r):
            if l > r:
                return [None]
            res = []
            for i in range(l, r + 1):
                left = build(l, i - 1)
                right = build(i + 1, r)
                for left_node in left:
                    for right_node in right:
                        res.append(TreeNode(i, left_node, right_node)) 
            return res

        return build(1, n)
```

## 最近公共祖先
#### [中]二叉搜索树的最近公共祖先
[https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-search-tree/](https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-search-tree/)

```
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, x):
#         self.val = x
#         self.left = None
#         self.right = None

class Solution(object):
    def lowestCommonAncestor(self, root, p, q):
        """
        :type root: TreeNode
        :type p: TreeNode
        :type q: TreeNode
        :rtype: TreeNode
        """
        while root:
            if q.val < root.val and p.val < root.val:
                root = root.left
            elif q.val > root.val and p.val > root.val:
                root = root.right
            else:
                return root
```
#### [中]二叉树的最近公共祖先
[https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/](https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/)

```
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, x):
#         self.val = x
#         self.left = None
#         self.right = None

class Solution(object):
    def lowestCommonAncestor(self, root, p, q):
        """
        :type root: TreeNode
        :type p: TreeNode
        :type q: TreeNode
        :rtype: TreeNode
        """
        if not root or root == p or root == q:
            return root
        left = self.lowestCommonAncestor(root.left, p, q)
        right = self.lowestCommonAncestor(root.right, p, q)
        if left and right:
            return root
        return left if left else right
```

满二叉树 节点数 = 2 ** 层高 - 1:

```
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution(object):
    def countNodes(self, root):
        """
        :type root: Optional[TreeNode]
        :rtype: int
        """
        def get_height(root):
            res = 0
            while root:
                root = root.left
                res += 1
            return res

        if not root:
            return 0
        
        l_height = get_height(root.left)
        r_height = get_height(root.right)

        if l_height == r_height:
            return 2 ** l_height + self.countNodes(root.right)
        else:
            return 2 ** r_height + self.countNodes(root.left)
```
## 二叉堆
#### [中]数组排序
[https://leetcode.cn/problems/sort-an-array/description/](https://leetcode.cn/problems/sort-an-array/description/)

|排序算法|平均时间复杂度|最坏时间复杂度|空间复杂度|稳定性|是否原地排序|
|-|-|-|-|-|-|
|快速排序|O(n log n)|O(n²)|O(log n)|不稳定|是|
|归并排序|O(n log n)|O(n log n)|O(n)|稳定|否|
|堆排序|O(n log n)|O(n log n)|O(1)|不稳定|是|

堆排序：

```
class Solution(object):
    def sortArray(self, nums):
        """
        :type nums: List[int]
        :rtype: List[int]
        """
        def sift_down(nums, i, size):
            """下沉操作，将堆顶元素下沉到正确的位置"""
            largest = i
            left = 2 * i + 1  # 二叉堆是完全二叉树
            right = 2 * i + 2

            if left < size and nums[largest] < nums[left]:
                largest = left
            if right < size and nums[largest] < nums[right]:
                largest = right
            if largest != i:
                nums[largest], nums[i] = nums[i], nums[largest]
                sift_down(nums, largest, size)
            
        def heapify(nums):
            for i in range(len(nums) // 2, -1, -1):
                sift_down(nums, i, len(nums))

        # 构建最大堆
        heapify(nums)

        # 堆排序
        for i in range(len(nums) - 1, 0, -1):
            nums[0], nums[i] = nums[i], nums[0]
            # 堆化剩下部分
            sift_down(nums, 0, i)
        return nums
```

```
class TreeNode(object):

    def __init__(self):
        self.child = {}
        self.is_end = False

class Trie(object):

    def __init__(self):
        self.root = TreeNode()
        
    def insert(self, word):
        """
        :type word: str
        :rtype: None
        """
        p = self.root
        for c in word:
            if c not in p.child:
                p.child[c] = TreeNode()
            p = p.child[c]
        p.is_end = True

    def search(self, word):
        """
        :type word: str
        :rtype: bool
        """
        p = self.root
        for c in word:
            if c not in p.child:
                return False
            p = p.child[c]
        return p.is_end
    
    def startsWith(self, prefix):
        """
        :type prefix: str
        :rtype: bool
        """
        p = self.root
        for c in prefix:
            if c not in p.child:
                return False
            p = p.child[c]
        return True
```

```
class TreeNode(object):

    def __init__(self):
        self.child = {}
        self.is_end = False

class WordDictionary(object):

    def __init__(self):
        self.root = TreeNode()
        

    def addWord(self, word):
        """
        :type word: str
        :rtype: None
        """
        p = self.root
        for c in word:
            if c not in p.child:
                p.child[c] = TreeNode()
            p = p.child[c]
        p.is_end = True
        

    def search(self, word):
        """
        :type word: str
        :rtype: bool
        """
        def dfs(index, node):
            if index == len(word):
                return node.is_end
            c = word[index]
            if c == '.':
                for child in node.child.values():
                    if dfs(index + 1, child):
                        return True
                return False
            if c not in node.child:
                return False
            return dfs(index + 1, node.child[c])
        return dfs(0, self.root)
```

```
class TreeNode(object):

    def __init__(self):
        self.child = {}
        self.is_end = False

class Solution(object):
    def replaceWords(self, dictionary, sentence):
        """
        :type dictionary: List[str]
        :type sentence: str
        :rtype: str
        """
        root = TreeNode()

        # 构建 Trie 树
        for word in dictionary:
            p = root
            for c in word:
                if c not in p.child:
                    p.child[c] = TreeNode()
                p = p.child[c]
            p.is_end = True

        def helper(word):
            """替换成前缀"""
            p = root
            prefix = ""
            for c in word:
                if c not in p.child:
                    return word  # 没匹配上任何 root
                p = p.child[c]
                prefix += c
                if p.is_end:
                    return prefix  # 找到最短 root
            return word  # 没有 root 是它的前缀

        # 替换句子
        words = sentence.split()
        res = [helper(word) for word in words]
        return " ".join(res)
        
```

```
class TreeNode(object):

    def __init__(self):
        self.child = {}
        self.is_end = False
        self.sum = 0

class MapSum(object):

    def __init__(self):
        self.root = TreeNode()
        self.info = {} 
        

    def insert(self, key, val):
        """
        :type key: str
        :type val: int
        :rtype: None
        """
        delta = val - self.info.get(key, 0)  # 差值
        self.info[key] = val

        p = self.root
        for c in key:
            if c not in p.child:
                p.child[c] = TreeNode()
            p = p.child[c]
            p.sum += delta  # 更新路径上所有前缀的和

        """
        :type prefix: str
        :rtype: int
        """
        p = self.root
        for c in prefix:
            if c not in p.child:
                return 0
            p = p.child[c]
        return p.sum
```
