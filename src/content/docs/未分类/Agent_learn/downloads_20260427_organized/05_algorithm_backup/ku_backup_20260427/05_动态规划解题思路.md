#### [易]斐波那契数
[https://leetcode.cn/problems/fibonacci-number/description/](https://leetcode.cn/problems/fibonacci-number/description/)

标准写法：

```
class Solution(object):
    def fib(self, n):
        """
        :type n: int
        :rtype: int
        """
        # 0 1 1 2 3
        # dp[i] = dp[i-1] + dp[i-2]
        if n == 0 or n == 1:
            return n
        
        dp = [0] * (n + 1)
        dp[0], dp[1] = 0, 1
        for i in range(2, n + 1):
            dp[i] = dp[i-1] + dp[i-2]
        return dp[n]
```

class Solution(object):

        """
        :type n: int
        :rtype: int
        """
        # 0 1 1 2 3
        if n == 0 or n == 1:
            return n

        l, r = 0, 1
        for i in range(2, n + 1):
            r, l = l + r, r
        return r

[https://leetcode.cn/problems/coin-change/](https://leetcode.cn/problems/coin-change/)

```
class Solution(object):
    def coinChange(self, coins, amount):
        """
        :type coins: List[int]
        :type amount: int
        :rtype: int
        """
        # dp[i] 表示组成金额 i 所需的最小硬币数
        # dp[i] = min(dp[i], 1 + dp[i-coin])
        dp = [float('inf')] * (amount + 1)
        dp[0] = 0
        for i in range(1, amount + 1):
            for coin in coins:
                if i >= coin:
                    dp[i] = min(dp[i], 1 + dp[i-coin])
        return dp[amount] if dp[amount] != float('inf') else -1
```

```
class Solution(object):
    def minFallingPathSum(self, matrix):
        """
        :type matrix: List[List[int]]
        :rtype: int
        """
        n = len(matrix)
        dp = matrix[0][:]

        for i in range(1, n):
            new_dp = [0] * n
            for j in range(n):
                mid = dp[j]
                left = dp[j-1] if j > 0 else float('inf')
                right = dp[j+1] if j < n - 1 else float('inf')
                new_dp[j] = matrix[i][j] + min(left, mid, right)
            dp = new_dp

        return min(dp)
```

```
class Solution(object):
    def minPathSum(self, grid):
        """
        :type grid: List[List[int]]
        :rtype: int
        """
        # dp[i][j]表示从(0,0)到位置(i,j)的最小路径和
        m, n = len(grid), len(grid[0])
        dp = [[0] * n for _ in range(m)]
        
        dp[0][0] = grid[0][0]
        
        # 第一列
        for i in range(1, m):
            dp[i][0] = dp[i-1][0] + grid[i][0]
            
        # 第一行
        for j in range(1, n):
            dp[0][j] = dp[0][j-1] + grid[0][j]
            
        for i in range(1, m):
            for j in range(1, n):
                dp[i][j] = grid[i][j] + min(dp[i-1][j], dp[i][j-1])
        
        return dp[-1][-1]
```
#### [中]K 站中转内最便宜的航班
[https://leetcode.cn/problems/cheapest-flights-within-k-stops/](https://leetcode.cn/problems/cheapest-flights-within-k-stops/)

```
class Solution(object):
    def findCheapestPrice(self, n, flights, src, dst, k):
        """
        :type n: int
        :type flights: List[List[int]]
        :type src: int
        :type dst: int
        :type k: int
        :rtype: int
        """
        # dp[i] 表示从 src 出发每个点的最小花费
        # 中转 k 次，限定最多可以走 k+1 条边
        INF = float('inf')
        dp = [INF] * n
        dp[src] = 0

        for _ in range(k + 1):
            new_dp = dp[:]
            for u, v, w in flights:
                if dp[u] != INF:
                    new_dp[v] = min(new_dp[v], dp[u] + w)
            dp = new_dp

        return -1 if dp[dst] == INF else dp[dst]
```

class Solution(object):

        :type s: str

class Solution(object):
    def maxSubArray(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # dp[i] 表示以 nums[i] 结尾的最大连续子数组的和
        # dp[i] = max(nums[i], dp[i-1] + nums[i])  ，优化空间只保留前一个 dp 值即可
        c_sum = max_sum = nums[0]
        for i in nums[1:]:
            c_sum = max(i, c_sum + i)
            max_sum = max(max_sum, c_sum)
        return max_sum

class Solution(object):
    def canPartition(self, nums):
        """
        :type nums: List[int]
        :rtype: bool
        """
        if sum(nums) % 2 != 0:
            return False

        target = sum(nums) // 2
        # dp[i] 表示是否可以找到和为 i 的子集
        dp = [False] * (target + 1)
        dp[0] = True
        for n in nums:
            for j in range(target, -1, -1):
                if j >= n:
                    dp[j] = dp[j] or dp[j - n]

        return dp[-1]
```

class Solution(object):
    def findTargetSumWays(self, nums, target):
        """
        :type nums: List[int]
        :type target: int
        :rtype: int
        """
        # 选中之和 - 未选中之和 = target
        # 选中之和 + 未选中之和 = add
        # 选中之和 = (add + target) // 2
        add = sum(nums)
        if add < abs(target) or (add + target) % 2 != 0:
            return 0
        target = (add + target) // 2
        dp = [0] * (target + 1)
        dp[0] = 1
        for n in nums:
            for j in range(target, -1, -1):
                if j >= n:
                    dp[j] = dp[j] + dp[j - n]
        return dp[-1]
```

class Solution(object):
    def predictTheWinner(self, nums):
        """
        :type nums: List[int]
        :rtype: bool
        """
        # dp[i][j]表示在 nums[i:j] 上，先手玩家能赢对手最多多少分
        n = len(nums)

        dp = [[0] * n for _ in range(n)]
    
        for i in range(n):
            dp[i][i] = nums[i]
    
        for l in range(2, n + 1):  
            for i in range(n - l + 1):
                j = i + l - 1
                dp[i][j] = max(
                    nums[i] - dp[i + 1][j],  # 拿左边，减去对手最优
                    nums[j] - dp[i][j - 1]  # 拿右边，减去对手最优
                )
    
        return dp[0][n - 1] >= 0
```

class Solution(object):
    def rob(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        def rob_linear(nums):
            prev, curr = 0, 0
            for num in nums:
                prev, curr = curr, max(curr, prev + num)
            return curr
            
        n = len(nums)
        if n == 0:
            return 0
        if n == 1:
            return nums[0]

        # 两种情况：不偷头 / 不偷尾
        return max(
            rob_linear(nums[1:]),
            rob_linear(nums[:-1])
        )

#### [中]打家劫舍 III
[https://leetcode.cn/problems/house-robber-iii/](https://leetcode.cn/problems/house-robber-iii/)

```
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution(object):
    def rob(self, root):
        """
        :type root: Optional[TreeNode]
        :rtype: int
        """
        def dfs(node):
            if not node:
                return (0, 0)

            left = dfs(node.left)
            right = dfs(node.right)

            rob_this = node.val + left[1] + right[1]  # 偷这个节点 → 不能偷左右
            not_rob_this = max(left) + max(right)  # 不偷这个节点 → 左右可以偷也可以不偷，取最大

            return (rob_this, not_rob_this)

        return max(dfs(root))
```
#### [中]删除并获得点数

```
class Solution(object):
    def deleteAndEarn(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # 类似于打家劫舍，选择了n就不能选择 n-1 和 n+1
        # dp[i] 表示考虑到数字 i 时的最大点数
        max_num = max(nums)
        points = [0] * (max_num + 1)
        for num in nums:
            points[num] += num

        dp = [0] * (max_num + 1)
        dp[0] = points[0]
        dp[1] = max(points[0], points[1])
        
        for i in range(2, max_num + 1):
            dp[i] = max(dp[i-1], dp[i-2] + points[i])

        return dp[max_num]
```

        # 今天持有股票的最大利润
        buy = [0] * n
        # 今天卖出股票后的最大利润
        sell = [0] * n
        # 今天处于冷冻期（不能买）的利润
        cool = [0] * n

        buy[0] = -prices[0]
        sell[0] = 0
        cool[0] = 0

        for i in range(1, n):
            buy[i] = max(buy[i-1], cool[i-1] - prices[i])
            sell[i] = buy[i-1] + prices[i]
            cool[i] = max(cool[i-1], sell[i-1])

        return max(sell[-1], cool[-1])
        
```

class Solution(object):
    def canCompleteCircuit(self, gas, cost):
        """
        :type gas: List[int]
        :type cost: List[int]
        :rtype: int
        """
        # tank 表示当前油箱剩下的油
        total, tank = 0, 0
        start = 0
        for i in range(len(gas)):
            total += gas[i] - cost[i]
            tank += gas[i] - cost[i]
            if tank < 0:
                tank = 0
                start = i + 1
        return start if total >= 0 else -1

class Solution(object):
    def jump(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # end 表示跳跃的最远边界
        res = 0
        end = 0
        max_reach = 0
        for i, n in enumerate(nums[:-1]):
            max_reach = max(max_reach, i + n)
            # 如果走到end，说明已经跳完了当前一步，必须增加跳跃次数，并更新end为能跳的最远地方
            if i == end:
                end = max_reach
                res += 1
        return res
```

class Solution(object):
    def eraseOverlapIntervals(self, intervals):
        """
        :type intervals: List[List[int]]
        :rtype: int
        """
        intervals.sort(key=lambda x: x[1])

        count = 0
        prev_end = float('-inf')

        for interval in intervals:
            start, end = interval
            if start >= prev_end:
                count += 1
                prev_end = end

        # 总区间数 - 保留区间数 = 需要移除的数量
        return len(intervals) - count
```

class Solution(object):

class Solution(object):

        :type clips: List[List[int]]
        :type time: int
        :rtype: int
        """

        next_end = 0
        count = 0
        i = 0 
        while end < time:

                next_end = max(next_end, clips[i][1])
                i += 1

            if next_end == end:
                return -1  # 无法继续扩展

            count += 1
            end = next_end
