## 拓扑排序
```
from collections import deque

def topological_sort(graph):
    # 统计每个节点的入度
    in_degree = {node: 0 for node in graph}
    for u in graph:
        for v in graph[u]:
            in_degree[v] += 1

    # 初始化队列，将所有入度为 0 的节点加入队列
    queue = deque([node for node in graph if in_degree[node] == 0])
    sorted_nodes = []  # 存储拓扑排序的结果

    # 开始拓扑排序
    while queue:
        u = queue.popleft()  # 取出一个入度为 0 的节点
        sorted_nodes.append(u)  # 加入排序结果

        # 遍历 u 的邻居，减少它们的入度
        for v in graph[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:  # 如果入度变为 0，加入队列
                queue.append(v)

    # 如果排序结果包含所有节点，则返回排序结果；否则说明图中有环
    if len(sorted_nodes) == len(graph):
        return sorted_nodes
    else:
        return "图中有环，无法进行拓扑排序！"

# 示例图（邻接表表示）
graph = {
    'A': ['B', 'C'],
    'B': ['D'],
    'C': ['D'],
    'D': ['E'],
    'E': []
}

# 进行拓扑排序
result = topological_sort(graph)

# 输出结果
print("拓扑排序结果:")
print(result)
# 拓扑排序结果:
# ['A', 'B', 'C', 'D', 'E']
```

## 二分图判定
* 使用 BFS 或 DFS 来遍历图。
* 对每个节点，给它染上颜色 0 或 1。
* 对于每个邻接的节点，确保它的颜色与当前节点不同。如果发现有邻接的节点颜色相同，则说明图不是二分图。
* 如果图是无向图，且所有节点都可以被成功染色，则该图是二分图。

```
from collections import deque

def isBipartite(graph):
    n = len(graph)
    color = [-1] * n  # -1表示未染色，0和1表示两种颜色
    
    for i in range(n):
        if color[i] == -1:  # 如果当前节点未染色
            queue = deque([i])
            color[i] = 0  # 初始染色为0
            
            while queue:
                node = queue.popleft()
                for neighbor in graph[node]:
                    if color[neighbor] == -1:  # 如果邻居未染色
                        color[neighbor] = color[node] ^ 1  # 染相反颜色
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:  # 如果邻居颜色相同
                        return False
    return True

# 示例用法
graph = [[1,3], [0,2], [1,3], [0,2]]
print(isBipartite(graph))  # 输出: True
```

## Dijkstra 算法
使用 Dijkstra 算法的前提，加权有向图，没有负权重边，求单源最短路径

```
import heapq

def dijkstra(graph, start):
    # 初始化距离字典，所有节点的距离设置为无穷大
    distances = {node: float('inf') for node in graph}
    distances[start] = 0  # 源节点到自身的距离为 0
    priority_queue = [(0, start)]  # 优先队列，存储 (距离, 节点)

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        # 如果当前距离大于已知距离，跳过
        if current_distance > distances[current_node]:
            continue

        # 遍历邻居节点
        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight

            # 如果找到更短的路径，更新距离并加入优先队列
            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(priority_queue, (distance, neighbor))

    return distances

# 示例图（邻接表表示）
graph = {
    'A': {'B': 1, 'C': 4},
    'B': {'A': 1, 'C': 2, 'D': 5},
    'C': {'A': 4, 'B': 2, 'D': 1},
    'D': {'B': 5, 'C': 1}
}

# 从节点 'A' 开始计算最短路径
start_node = 'A'
shortest_paths = dijkstra(graph, start_node)

# 输出结果
print(f"从节点 {start_node} 到其他节点的最短路径:")
for node, distance in shortest_paths.items():
    print(f"到节点 {node} 的最短距离: {distance}")
    
# 从节点 A 到其他节点的最短路径:
# 到节点 A 的最短距离: 0
# 到节点 B 的最短距离: 1
# 到节点 C 的最短距离: 3
# 到节点 D 的最短距离: 4
```

class Solution(object):
    def networkDelayTime(self, times, n, k):
        """
        :type times: List[List[int]]
        :type n: int
        :type k: int
        :rtype: int
        """
        # 邻接表
        graph = {i:[] for i in range(n)}
        for i, j, w in times:
            graph[i-1].append((j-1, w))

        # 优先级队列
        dist = {i:float('inf') for i in range(n)}
        k = k-1
        dist[k] = 0
        priority_queue = [(0, k)]
        while priority_queue:
            cur_dist, cur_node = heapq.heappop(priority_queue)
            if cur_dist > dist[cur_node]: 
                continue
            for node, wt in graph[cur_node]:
                new_dist = cur_dist + wt
                if new_dist < dist[node]:
                    dist[node] = new_dist
                    heapq.heappush(priority_queue, (new_dist, node))

        res = max(dist.values())
        return res if res != float('inf') else -1

class Solution(object):
    def minimumEffortPath(self, heights):
        """
        :type heights: List[List[int]]
        :rtype: int
        """
        m, n = len(heights), len(heights[0])
        def adj(height, i, j):
            """i, j邻居 距离差"""
            res = []
            poses = [(-1,0),(1,0),(0,-1),(0,1)] # 上下左右
            for pos in poses:
                ni, nj = i + pos[0], j + pos[1]
                if 0 <= ni < m and 0 <= nj < n:
                    wt = abs(height[ni][nj] - height[i][j])
                    res.append((ni, nj, wt))
            return res
        # m * n 矩阵
        dist = [[float('inf')] * n for i in range(m)]
        dist[0][0] = 0
        priority_queue = [(0, 0, 0)] 
        while priority_queue:
            cur_dist, cur_i, cur_j = heapq.heappop(priority_queue)
            if cur_dist > dist[cur_i][cur_j]:
                continue
            for node_i, node_j, wt in adj(heights, cur_i, cur_j):
                new_dist = max(cur_dist, wt)
                if new_dist < dist[node_i][node_j]:
                    dist[node_i][node_j] = new_dist
                    heapq.heappush(priority_queue, (new_dist, node_i, node_j))
        return dist[m-1][n-1]

## Prim 最小生成树
```
import heapq

def prim(graph, start):
    # 初始化
    mst = []  # 存储最小生成树的边
    visited = set()  # 记录已访问的节点
    priority_queue = []  # 优先队列，存储 (权重, 当前节点, 父节点)

    # 从起始节点开始
    heapq.heappush(priority_queue, (0, start, None))  # (权重, 当前节点, 父节点)

    while priority_queue:
        weight, current_node, parent = heapq.heappop(priority_queue)

        # 如果节点已访问，跳过
        if current_node in visited:
            continue

        # 标记节点为已访问
        visited.add(current_node)

        # 如果父节点不为空，将边加入最小生成树
        if parent is not None:
            mst.append((parent, current_node, weight))

        # 遍历当前节点的邻居
        for neighbor, edge_weight in graph[current_node].items():
            if neighbor not in visited:
                heapq.heappush(priority_queue, (edge_weight, neighbor, current_node))

    return mst

# 示例图（邻接表表示）
graph = {
    'A': {'B': 1, 'C': 4},
    'B': {'A': 1, 'C': 2, 'D': 5},
    'C': {'A': 4, 'B': 2, 'D': 1},
    'D': {'B': 5, 'C': 1}
}

# 从节点 'A' 开始计算最小生成树
start_node = 'A'
mst = prim(graph, start_node)

# 输出结果
print("最小生成树的边:")
for u, v, w in mst:
    print(f"边 {u} - {v}, 权重: {w}")
# 最小生成树的边:
# 边 A - B, 权重: 1
# 边 B - C, 权重: 2
# 边 C - D, 权重: 1
```

```
class Solution(object):
    def minCostConnectPoints(self, points):
        """
        :type points: List[List[int]]
        :rtype: int
        """
        n = len(points)
        visited = set()
        min_heap = [(0, 0)] 
        total_cost = 0

        while len(visited) < n:
            cost, u = heapq.heappop(min_heap)
            if u in visited:
                continue
            visited.add(u)
            total_cost += cost

            for v in range(n):
                if v not in visited:
                    dist = abs(points[u][0] - points[v][0]) + abs(points[u][1] - points[v][1])
                    heapq.heappush(min_heap, (dist, v))

        return total_cost

## 并查集
```
class DSU:
    def __init__(self, n):
        self.parent = [i for i in range(n)]
        self.rank = [0] * n
        self.count = n  # 初始连通分量数

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # 路径压缩
        return self.parent[x]

    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return  # 已连通
        # 按秩合并
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1
        self.count -= 1  # 连通分量减少

    def is_connected(self, x, y):
        return self.find(x) == self.find(y)

    def get_count(self):
        return self.count  # O(1) 查询连通分量数


# 创建一个大小为5的并查集
dsu = DSU(5)
print(dsu.get_count())  # 5

# 初始状态：每个元素都是自己的父节点
print(dsu.parent)  # [0, 1, 2, 3, 4]
print(dsu.rank)    # [0, 0, 0, 0, 0]

# 合并操作
dsu.union(0, 1)
print(dsu.parent)  # [0, 0, 2, 3, 4]
print(dsu.rank)    # [1, 0, 0, 0, 0]
dsu.union(3, 2)
print(dsu.parent)  # [0, 0, 3, 3, 4]
print(dsu.rank)    # [1, 0, 0, 1, 0]
print(dsu.get_count())  # 3

# 检查连接
print(dsu.is_connected(0, 1))  # True
print(dsu.is_connected(0, 2))  # False

# 再次合并
dsu.union(1, 3)
print(dsu.parent)  # [0, 0, 3, 0, 4]
print(dsu.rank)    # [2, 0, 0, 1, 0]
print(dsu.get_count())  # 2
print(dsu.is_connected(0, 2))  # True
print(dsu.is_connected(0, 4))  # False

# 查询连通分量数
print(dsu.get_count())  # 2

## Kruskal 最小生成树
```
def find(parent, x):
    """
    查找节点 x 的根节点，并进行路径压缩。
    :param parent: 并查集的父节点字典。
    :param x: 要查找的节点。
    :return: 节点 x 的根节点。
    """
    if parent[x] != x:  # 如果 x 不是根节点
        parent[x] = find(parent, parent[x])  # 递归查找根节点，并进行路径压缩
    return parent[x]  # 返回根节点


def kruskal(graph):
    """
    Kruskal 算法求解最小生成树。
    :param graph: 图的邻接表表示，格式为 {u: {v: weight, ...}, ...}。
    :return: 最小生成树的边列表，格式为 [(u, v, weight), ...]。
    """
    # 获取所有节点和边
    nodes = list(graph.keys())  # 所有节点的列表
    edges = []  # 存储所有边的列表
    for u in graph:
        for v, w in graph[u].items():
            edges.append((w, u, v))  # 存储边 (权重, 节点1, 节点2)

    # 按边的权重从小到大排序
    edges.sort()  # 根据权重排序

    # 初始化并查集
    parent = {node: node for node in nodes}  # 每个节点的父节点初始化为自己
    rank = {node: 1 for node in nodes}  # 每个节点的秩初始化为 1
    mst = []  # 存储最小生成树的边

    # 遍历所有边
    for w, u, v in edges:
        # 查找节点 u 和 v 的根节点
        root_x = find(parent, u)
        root_y = find(parent, v)

        # 如果两个节点不在同一个集合中，则加入最小生成树
        if root_x != root_y:
            # 按秩合并
            if rank[root_x] > rank[root_y]:
                parent[root_y] = root_x  # 将秩较小的树合并到秩较大的树
            elif rank[root_x] < rank[root_y]:
                parent[root_x] = root_y
            else:
                parent[root_y] = root_x  # 如果秩相等，则合并并增加秩
                rank[root_x] += 1

            # 将边加入最小生成树
            mst.append((u, v, w))
            
    print("rank: %s" % rank)
    print("parent: %s" % parent)
    return mst 


# 示例图（邻接表表示）
graph = {
    'A': {'B': 1, 'C': 4},
    'B': {'A': 1, 'C': 2, 'D': 5},
    'C': {'A': 4, 'B': 2, 'D': 1},
    'D': {'B': 5, 'C': 1}
}

# 计算最小生成树
mst = kruskal(graph)

# 输出结果
print("最小生成树的边:")
for u, v, w in mst:
    print(f"边 {u} - {v}, 权重: {w}")

# rank: {'A': 3, 'B': 1, 'C': 2, 'D': 1}
# parent: {'A': 'A', 'B': 'A', 'C': 'A', 'D': 'A'}
# 最小生成树的边:
# 边 A - B, 权重: 1
# 边 C - D, 权重: 1
# 边 B - C, 权重: 2
```
