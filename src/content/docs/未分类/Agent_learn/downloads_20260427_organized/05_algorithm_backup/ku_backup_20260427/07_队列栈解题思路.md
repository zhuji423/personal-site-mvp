class MinStack(object):

    def __init__(self):
        self.data = []
        self.stack = []
        
    def push(self, val):
        """
        :type val: int
        :rtype: None
        """
        self.data.append(val)
        if not self.stack or self.stack[-1] >= val:
            self.stack.append(val)

    def pop(self):
        """
        :rtype: None
        """
        cur = self.data.pop()
        if cur == self.stack[-1]:
            self.stack.pop()

    def top(self):
        """
        :rtype: int
        """
        return self.data[-1]

    def getMin(self):
        """
        :rtype: int
        """
        return self.stack[-1]
```

class Solution(object):
    def simplifyPath(self, path):
        """
        :type path: str
        :rtype: str
        """
        parts = path.split("/")
        stack = []
        for c in parts:
            if c == '' or c == '.':
                continue
            if c == '..':
                if stack:
                    stack.pop()
                continue
            stack.append(c)
        return '/' + '/'.join(list(stack))

```
# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution(object):
    def reorderList(self, head):
        """
        :type head: Optional[ListNode]
        :rtype: None Do not return anything, modify head in-place instead.
        """
        stack = []
        p = head 
        while p:
            stack.append(p)
            p = p.next
        p = head
        while p:
            last = stack.pop()
            next_p = p.next
            if last == next_p or last.next == next_p:
                last.next = None    
                break
            p.next = last
            last.next = next_p
            p = next_p
        return head
```

class Solution(object):

        :type tokens: List[str]
        :rtype: int
        """

            if c == '+':
                return i + j

                return i - j

```
class Solution(object):
    def lengthLongestPath(self, input):
        """
        :type input: str
        :rtype: int
        """
        stack = []
        max_len = 0
        for part in input.split('\n'):
            level = part.rfind('\t') + 1
            while level < len(stack):
                stack.pop()
            if '.' in part:
                max_len = max(max_len, sum(stack) + len(stack) + len(part) - level)
            stack.append(len(part) - level)
        return max_len
```
#### [中]去除重复字母
[https://leetcode.cn/problems/remove-duplicate-letters/](https://leetcode.cn/problems/remove-duplicate-letters/)

```
class Solution(object):
    def removeDuplicateLetters(self, s):
        """
        :type s: str
        :rtype: str
        """
        # 统计字符最后出现的位置
        info = {c: i for i, c in enumerate(s)}
        stack = []
        for i, c in enumerate(s):
            if c in stack:
                continue
            # 当前字符比栈顶小且栈顶字符后续还会出现
            while stack and stack[-1] > c and info[stack[-1]] > i:
                stack.pop()
            stack.append(c)
        return ''.join(stack)
```

class LRUCache(object):

    def __init__(self, capacity):
        """
        :type capacity: int
        """
        self.capacity = capacity
        self.data = {}
        self.ord = deque()

    def get(self, key):
        """
        :type key: int
        :rtype: int
        """
        if key not in self.data:
            return -1
        self.ord.remove(key)
        self.ord.append(key)
        return self.data.get(key)


    def put(self, key, value):
        """
        :type key: int
        :type value: int
        :rtype: None
        """
        if key not in self.data and len(self.data) == self.capacity:
                # 满了
                del_key = self.ord.popleft()
                del self.data[del_key]
        
        if key in self.data:
            self.ord.remove(key)
        self.data[key] = value
        self.ord.append(key)

class MyCircularQueue(object):

    def __init__(self, k):
        """
        :type k: int
        """
        self.data = [None] * k
        self.l = 0
        self.r = 0  # 指向尾的下一个
        self.size = 0

    def _resize(self, new_size):
        new_data = [None] * new_size
        for i in range(self.size):
            new_data[i] = self.data[(self.l + i) % self.size]
        self.l = 0
        self.r = self.size
        self.data = new_data
        

    def enQueue(self, value):
        """
        :type value: int
        :rtype: bool
        """
        # if self.size = len(self.data):
        #     self._resize(2 * self.size)
        if self.size == len(self.data):
            return False

        self.data[self.r] = value
        self.r += 1
        if self.r == len(self.data):
            self.r = 0

        self.size += 1
        return True

    def deQueue(self):
        """
        :rtype: bool
        """
        if self.size == 0:
            return False

        # if self.size == len(self.data) // 4:
        #     self.resize(len(self.data) // 2)

        self.data[self.l] = None
        self.l += 1
        if self.l == len(self.data):
            self.l = 0

        self.size -= 1
        return True
        

    def Front(self):
        """
        :rtype: int
        """
        if self.size == 0:
            return -1
        return self.data[self.l]
        

    def Rear(self):
        """
        :rtype: int
        """
        if self.size == 0:
            return -1
        return self.data[self.r - 1] if self.r > 0 else self.data[-1]
        

    def isEmpty(self):
        """
        :rtype: bool
        """
        return self.size == 0
        

    def isFull(self):
        """
        :rtype: bool
        """
        return self.size == len(self.data)
```
## 括号类问题
#### [易]**有效的括号**
[https://leetcode.cn/problems/valid-parentheses/](https://leetcode.cn/problems/valid-parentheses/)

```
class Solution(object):
    def isValid(self, s):
        """
        :type s: str
        :rtype: bool
        """
        def is_ok(i, j):
            return i=='{' and j=='}' or i=='[' and j==']' or i=='(' and j==')'

        stack = []
        for c in s:
            if stack and is_ok(stack[-1], c):
                stack.pop()
                continue
            stack.append(c)
        return len(stack) == 0
```

[https://leetcode.cn/problems/next-greater-element-i/](https://leetcode.cn/problems/next-greater-element-i/)

```
class Solution(object):
    def nextGreaterElement(self, nums1, nums2):
        """
        :type nums1: List[int]
        :type nums2: List[int]
        :rtype: List[int]
        """
        stack = []
        info = {}
        for i, c in enumerate(nums2):
            while stack and c > stack[-1]:
                info[stack[-1]] = c
                stack.pop()
            stack.append(c)
        return [info[i] if i in info else -1 for i in nums1]
```

class Solution(object):
    def nextGreaterElements(self, nums):
        """
        :type nums: List[int]
        :rtype: List[int]
        """
        n = len(nums)
        info = {}
        stack = []
        for i in range(n * 2):
            idx = i % n
            while stack and nums[idx] > nums[stack[-1]]:
                info[stack.pop()] = nums[idx]
            if i < n:
                stack.append(idx)

        return [info.get(i, -1) for i in range(n)]

class Solution(object):
    def removeKdigits(self, num, k):
        """
        :type num: str
        :type k: int
        :rtype: str
        """
        stack = []
        for c in num:
            while k > 0 and stack and c < stack[-1]:
                stack.pop()
                k -= 1
            stack.append(c)

        while k > 0:
            stack.pop()
            k -= 1

        res = ''.join(stack).lstrip('0')
        return res if res else '0'

#### [难]滑动窗口最大值
[https://leetcode.cn/problems/sliding-window-maximum/](https://leetcode.cn/problems/sliding-window-maximum/)

```
class Solution(object):
    def maxSlidingWindow(self, nums, k):
        """
        :type nums: List[int]
        :type k: int
        :rtype: List[int]
        """
        maxq = deque()
        res = []
        for i, c in enumerate(nums):
            if i < k - 1:
                # push c
                while maxq and maxq[-1] < c:
                    maxq.pop()
                maxq.append(c)
                continue

            # push c
            while maxq and maxq[-1] < c:
                maxq.pop()
            maxq.append(c)
            
            # 更新max
            max_val = maxq[0]
            res.append(max_val)
            
            # pop nums[i - k + 1]
            if maxq[0] == nums[i - k + 1]:
                maxq.popleft()
        return res
   
```

class Checkout(object):

    def __init__(self):
        self.q = deque()
        self.maxq = deque()
        
    def get_max(self):
        """
        :rtype: int
        """
        if len(self.q) == 0:
            return -1
        return self.maxq[0]
        

    def add(self, value):
        """
        :type value: int
        :rtype: None
        """
        self.q.append(value)
        while self.maxq and self.maxq[-1] < value:
            self.maxq.pop()
        self.maxq.append(value)
        
    def remove(self):
        """
        :rtype: int
        """
        if len(self.q) == 0:
            return -1
        del_val = self.q.popleft()
        if del_val == self.maxq[0]:
            self.maxq.popleft()
        return del_val

#### [中]**绝对差不超过限制的最长连续子数组**
[https://leetcode.cn/problems/longest-continuous-subarray-with-absolute-diff-less-than-or-equal-to-limit/](https://leetcode.cn/problems/longest-continuous-subarray-with-absolute-diff-less-than-or-equal-to-limit/)

```
class Solution(object):
    def longestSubarray(self, nums, limit):
        """
        :type nums: List[int]
        :type limit: int
        :rtype: int
        """
        maxq = deque()  # 递减队列，队首是当前窗口最大值
        minq = deque()  # 递增队列，队首是当前窗口最小值
        # 滑动窗口, 只要 max_q[0] - min_q[0] > limit，就收缩左边界
        l, r = 0, 0
        res = 0
        while r < len(nums):
            # push right
            right = nums[r]
            while maxq and maxq[-1] < right:
                maxq.pop()
            maxq.append(right)

            while minq and minq[-1] > right:
                minq.pop()
            minq.append(right)

            while maxq[0] - minq[0] > limit:
                # pop left
                left = nums[l]
                if maxq[0] == left:
                    maxq.popleft()
                if minq[0] == left:
                    minq.popleft()
                l += 1
            
            res = max(res, r - l + 1)
            r += 1
        return res
```
